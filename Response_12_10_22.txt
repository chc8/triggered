    def accept_clients(self):
        while True:

            threading.Thread(target=self.handle_client, args=(conn,), daemon=True).start()

class Tumbleweed:
    def __init__(self, screen_w, screen_y):
        self.x = -100
        self.y = screen_y
        self.angle = 0
        self.screen_w = screen_w

    def update(self):
        self.x += 5
        self.angle += 10
        self.y += math.sin(self.x * 0.05) * 5

    def draw(self, screen):
        color = (180, 140, 80)
        cx, cy = int(self.x), int(self.y)
        for i in range(8):
            a = math.radians(self.angle + (i * 45))
            ex = cx + math.cos(a) * 30
            ey = cy + math.sin(a) * 30
            pygame.draw.line(screen, color, (cx, cy), (ex, ey), 2)
        pygame.draw.circle(screen, color, (cx, cy), 25, 2)

class TriggeredGameApp:

        # Biologically outside 2 standard deviations roughly

        elif event.key == pygame.K_RETURN:
            step = self.steps[self.setup_step]
            val = self.input_text.strip()
            
            try:
                if step == 'MODE':
                    if val.upper() in ['S', 'C']:
                        self.params[step] = val.upper()
                        self.is_host = (val.upper() == 'S')
                        self.setup_step += 1
                    else: self.error = "S for Host, C for Client"
                elif step in ['AGE', 'HEIGHT', 'WEIGHT', 'DECKS', 'PORT']:
                    num = float(val) if step == 'HEIGHT' else int(val)
                    if num <= 0: raise ValueError
                    
                    if step == 'PORT':
                        if num < 1024: self.error = "Port reserved! Use 1024-65535"
                        elif num > 65535: self.error = "Port too high"
                        else: 
                            self.params[step] = num
                            self.finalize()
                    elif step == 'DECKS':
                        self.params[step] = num
                        self.setup_step += 1
                    else:
                        self.params[step] = num
                        self.setup_step += 1
                else:
                    self.params[step] = val
                    self.setup_step += 1
                self.input_text = ""
                self.error = ""
            except:
                self.error = "Must be a positive number!"

    def finalize(self):
        # Start server/client logic here
        self.state = 'GAME'

    def run(self):
        clock = pygame.time.Clock()
        while True:
            for event in pygame.event.get():
                if event.type == pygame.QUIT: pygame.quit(); sys.exit()
                if event.type == pygame.KEYDOWN and self.state == 'SETUP':

                if self.error:
                    err = self.font.render(self.error, True, (255, 100, 100))
                    self.screen.blit(err, (100, 350))

            pygame.display.flip()
            clock.tick(60)

if __name__ == "__main__":
