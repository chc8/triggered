            self.broadcast({'msg': f"Cards dealt face down in the dirt. The Bullseye is:"})
            self.broadcast({'msg': f"\n      >>> {bullseye['value']} of {bullseye['suit']} <<<\n"})
            
            self.safe_sleep(2)
            
            with self.lock:
                self.trigger_calls.clear()
                self.trigger_phase = True
                
            self.broadcast({'msg': f"{dealer['name']} draws and yells: 'CLICK CLICK'! (Turn 'em over!)"})
            
            cards_display = "Cards revealed on the table:\n" + "\n".join([f"  - {name}: {c['value']} of {c['suit']}" for name, c in round_cards.items()])
            self.broadcast({'msg': cards_display})
            
            match_exists = any(c['value'] == bullseye['value'] for c in round_cards.values())
            pot = len(self.players) + 1
            winner = None

            if match_exists:
                for p in self.players:
                    if p.get('is_ai'):
                        if random.random() < 0.15: 
                            reaction = random.uniform(0.15, 0.7)
                        else:
                            reaction = random.uniform(0.8, 4.8)
                        threading.Timer(reaction, self.register_trigger, args=(p['name'],)).start()

                self.safe_sleep(TRIGGER_WINDOW)
                
                initiate_quickdraw = False
                qd_ties = []
                
                with self.lock:
                    self.trigger_phase = False 
                    
                    if self.trigger_calls:
                        self.trigger_calls.sort(key=lambda x: x[0])
                        fastest_time = self.trigger_calls[0][0]
                        ties = [name for t, name in self.trigger_calls if t - fastest_time < 0.1]
                        
                        if len(ties) > 1:
                            initiate_quickdraw = True
                            qd_ties = ties
                        else:
                            winner_name = ties[0]
                            self.broadcast({'msg': f"*** {winner_name} WAS THE FASTEST GUN! ***"})
                            winner = next(p for p in self.players if p['name'] == winner_name)
                    else:
                        self.broadcast({'msg': "Nobody pulled the trigger! Let's see who's holdin' the high card..."})
                        high_players = self.get_high_card_winners(round_cards)
                        if len(high_players) > 1:
                            initiate_quickdraw = True
                            qd_ties = high_players
                        else:
                            winner_name = high_players[0]
                            self.broadcast({'msg': f"{winner_name} holds the highest card!"})
                            winner = next(p for p in self.players if p['name'] == winner_name)

                if initiate_quickdraw:
                    self.broadcast({'msg': f"Trigger TIE between {', '.join(qd_ties)}! Initiating Quickdraw Round!"})
                    winner, extra_pot = self.play_quickdraw_round(qd_ties, dealer, deck, pot)
                    pot += extra_pot

            else:
                for p in self.players:
                    if p.get('is_ai'):
                        if random.random() < 0.02: 
                            reaction = random.uniform(0.8, 3.5)
                            threading.Timer(reaction, self.register_trigger, args=(p['name'],)).start()

                self.safe_sleep(3.0) 
                
                initiate_quickdraw = False
                qd_ties = []
                
                with self.lock:
                    self.trigger_phase = False
                    if self.trigger_calls:
                        self.trigger_calls.sort(key=lambda x: x[0])
                        first_trigger_name = self.trigger_calls[0][1]
                        
                        if first_trigger_name == dealer['name']:
                            self.broadcast({'msg': f"Misfire! The Dealer ({dealer['name']}) drew on a ghost!"})
                            self.broadcast({'msg': f"The dealer loses! {pot} cards are scattered randomly to the other players."})
                            other_players = [p for p in self.players if p['name'] != dealer['name']]
                            if other_players:
                                for _ in range(pot):
                                    random.choice(other_players)['score'] += 1
                            winner = None 
                        else:
                            self.broadcast({'msg': f"Misfire! {first_trigger_name} got trigger-happy with no match."})
                            self.broadcast({'msg': f"Dealer {dealer['name']} takes the pot."})
                            winner = dealer
                    else:
                        high_players = self.get_high_card_winners(round_cards)
                        if len(high_players) > 1:
                            initiate_quickdraw = True
                            qd_ties = high_players
                        else:
                            winner_name = high_players[0]
                            self.broadcast({'msg': f"{winner_name} holds the highest card!"})
                            winner = next((p for p in self.players if p['name'] == winner_name), None)

                if initiate_quickdraw:
                    self.broadcast({'msg': f"Tie for the high card between {', '.join(qd_ties)}! Initiating Quickdraw Round!"})
                    winner, extra_pot = self.play_quickdraw_round(qd_ties, dealer, deck, pot)
                    pot += extra_pot

            if winner:
                winner['score'] += pot
                self.broadcast({'msg': f"{winner['name']} wins the round and collects {pot} cards!\n"})
            else:
                self.broadcast({'msg': "The round ends in chaos! Cards distributed.\n"})
            
            self.broadcast_scores()
            self.round_num += 1
            dealer_idx = (dealer_idx + 1) % len(self.players)
            self.safe_sleep(2.5)

        self.broadcast({'msg': "\n=== GAME OVER! The deck ran dry. ==="})
        remaining = len(deck)
        
        # Failsafe if the original dealer was kicked out mid-game
        first_dealer_obj = next((p for p in self.players if p['name'] == original_dealer), None)
        if first_dealer_obj:
            first_dealer_obj['score'] += remaining
            self.broadcast({'msg': f"First dealer {original_dealer} pockets the remaining {remaining} cards."})
            
        self.broadcast_scores()
        self.safe_sleep(2.0)
        
        if self.players:
            # Check for Moonshine Shootout
            high_score = max(p['score'] for p in self.players)
            high_players = [p for p in self.players if p['score'] == high_score]
            zero_players = [p for p in self.players if p['score'] == 0]
            
            if high_score > 0 and len(zero_players) > 0:
                self.moonshine_shootout(high_players, zero_players)
                self.safe_sleep(3.0)
            
            self.players.sort(key=lambda x: x['score'], reverse=True)
            standings = "\nFINAL BOUNTIES (Scores):\n" + "\n".join([f"  {p['name']}: {p['score']} cards" for p in self.players])
            self.broadcast({'msg': standings})
            
        self.game_active = False
        
        # FINAL BOUNTY BOARD UPDATE
        self.broadcast_scores() 
        self.broadcast({'msg': "\nHost can press 'R' to play a new game, or close the window to exit."})

        self.shake_timer = 0
        self.flash_alpha = 0

    def update(self):

        self.y = lerp(self.y, self.target_y, 0.1)
        
        self.draw_x = self.x
        self.draw_y = self.y

            self.draw_x += random.randint(-5, 5)
            self.draw_y += random.randint(-5, 5)
            self.shake_timer -= 1
            
        if self.flash_alpha > 0:
            self.flash_alpha = max(0, self.flash_alpha - 8)

        except ValueError: pass

                if self.tumbleweed_state == 'OFF':
                    self.reset_tumbleweeds()
                self.tumbleweed_state = 'ROLLING'
        else:
            if self.tumbleweed_state == 'ROLLING':
                self.tumbleweed_state = 'BLOWING_AWAY'

    def validate_field(self, step_id, val):
        try:
            if step_id == 'MODE':
                if val.lower() not in ['s', 'c']: return False, "Type 'S' for Host or 'C' for Client."
                self.is_host = (val.lower() == 's')
                
                if self.is_host:
                    self.setup_sequence = ['MODE', 'NAME', 'AGE', 'HEIGHT', 'WEIGHT', 'DECKS', 'NETWORK', 'PORT']
                else:
                    self.setup_sequence = ['MODE', 'NAME', 'AGE', 'HEIGHT', 'WEIGHT', 'NETWORK', 'PORT']
                    
            elif step_id == 'NAME':
                if not val.strip(): return False, "Name cannot be empty."
                self.name = val.strip()
            elif step_id == 'AGE':
                v = float(val)
                if v <= 0: return False, "Age must be > 0."
                self.age = v
            elif step_id == 'HEIGHT':
                v = float(val)
                if v <= 0: return False, "Height must be > 0."
                self.height = v
            elif step_id == 'WEIGHT':
                v = float(val)
                if v <= 0: return False, "Weight must be > 0."
                self.weight_lbs = v
                self.thumper = calculate_thumper(self.age, self.height, self.weight_lbs * 0.453592)
            elif step_id == 'DECKS':
                v = int(val)
                if not (1 <= v <= 5): return False, "Decks must be 1-5."
                self.num_decks = v
            elif step_id == 'NETWORK':
                if self.is_host:
                    v = int(val)
                    if not (0 <= v <= 7): return False, "AI count 0-7."
                    self.ai_count = v
                else:
                    if not val.strip(): return False, "IP cannot be empty."
                    self.host_ip = val.strip()
            elif step_id == 'PORT':
                v = int(val)
                if not (1024 < v <= 65535): return False, "Port 1025-65535."
                self.port = v
            return True, ""
        except ValueError:
            return False, "Invalid format. Numbers required here."

    def handle_setup_input(self, event):
        current_field = self.setup_sequence[self.setup_idx]
        
        if event.key == pygame.K_BACKSPACE:
            self.setup_data[current_field] = self.setup_data[current_field][:-1]
            self.setup_error = ""
        elif event.key == pygame.K_UP:
            if self.setup_idx > 0:
                self.setup_idx -= 1
                self.setup_error = ""
                self.evaluate_tumbleweeds()
        elif event.key == pygame.K_DOWN:
            valid, err = self.validate_field(current_field, self.setup_data[current_field])
            if valid:
                if self.setup_idx < len(self.setup_sequence) - 1:
                    self.setup_idx += 1
                    self.setup_error = ""
                    self.evaluate_tumbleweeds()
            else:
                self.setup_error = err
        elif event.key in (pygame.K_RETURN, pygame.K_KP_ENTER):
            valid, err = self.validate_field(current_field, self.setup_data[current_field])
            if valid:
                if self.setup_idx < len(self.setup_sequence) - 1:
                    self.setup_idx += 1
                    self.setup_error = ""
                    self.evaluate_tumbleweeds()
                else:
                    all_valid = True
                    for i, f in enumerate(self.setup_sequence):
                        v, e = self.validate_field(f, self.setup_data[f])
                        if not v:
                            self.setup_idx = i
                            self.setup_error = e
                            all_valid = False
                            break
                    if all_valid:
                        self.finalize_setup()
            else:
                self.setup_error = err
        else:
            if event.unicode.isprintable() and len(self.setup_data[current_field]) < 20:
                self.setup_data[current_field] += event.unicode
                self.setup_error = ""

    def finalize_setup(self):
        pygame.display.set_caption("TRIGGERED - High Res Edition" + (" [HOST]" if self.is_host else ""))
        
        if self.is_host:
            # Bypass the loop validator to guarantee the AI count variable is rigidly populated
            try:
                ai_qty = int(self.setup_data.get('NETWORK', '0'))
            except:
                ai_qty = 0
            self.ai_count = ai_qty
            
            self.server = TriggeredServer(self.port, self.ai_count, self.num_decks)
            self.server.start()
            time.sleep(0.5)
            self.host_ip = '127.0.0.1'
            
        self.connect_to_server()

    def send_heartbeats(self):
        """ Keeps the client connection alive by pinging the server """
        while self.running and self.client_socket:
            try:
                msg = json.dumps({'action': 'ping'}) + "\n"
                self.client_socket.sendall(msg.encode())
            except:
                pass
            time.sleep(5)

    def connect_to_server(self):
        self.client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        try:
            self.client_socket.connect((self.host_ip, self.port))
            info = json.dumps({'name': self.name, 'thumper': self.thumper})
            self.client_socket.sendall(info.encode())
            
            self.state = 'TITLE'
            
            self.add_log_message("Welcome to the Saloon!", SVGA['TEXT_LIGHT'])
            self.log_game_text("Welcome to the Saloon!")
            
            if self.is_host:
                self.add_log_message("You are the Host. Press 'S' to Start.", SVGA['GOLD'])
                self.log_game_text("You are the Host. Press 'S' to Start.")
            else:
                self.add_log_message("Waiting for the Host to start...", SVGA['TEXT_LIGHT'])
                self.log_game_text("Waiting for the Host to start...")
                
            self.add_log_message("Press 'T' to pull the TRIGGER!", SVGA['TEXT_LIGHT'])
            self.log_game_text("Press 'T' to pull the TRIGGER!")
            self.add_log_message("Press 'Q' to toggle audio.", SVGA['TEXT_LIGHT'])
            self.log_game_text("Press 'Q' to toggle audio.")
            self.add_log_message("Press 'P' to Pause.", SVGA['TEXT_LIGHT'])
            self.log_game_text("Press 'P' to Pause.")
            
            threading.Thread(target=self.receive_messages, daemon=True).start()
            threading.Thread(target=self.send_heartbeats, daemon=True).start()
        except ConnectionRefusedError:
            self.setup_idx = self.setup_sequence.index('PORT')
            self.setup_error = "Connection Failed. Check IP/Port and try again."

    def receive_messages(self):
        buffer = ""
        while self.running and self.state in ['TITLE', 'DOOR_ANIM', 'GAME']:
            try:
                data = self.client_socket.recv(2048).decode()
                if not data: break
                buffer += data
                
                while "\n" in buffer:
                    line, buffer = buffer.split("\n", 1)
                    line = line.strip()
                    if not line: continue
                    
                    try:
                        msg = json.loads(line)
                        
                        if isinstance(msg, dict):
                            if msg.get('action') == 'update_scores':
                                self.current_scores = msg.get('scores', {})
                                continue
                                
                            if msg.get('action') == 'remove_player':
                                name_to_remove = msg.get('name')
                                if name_to_remove in self.player_cards:
                                    del self.player_cards[name_to_remove]
                                if name_to_remove in self.all_player_names:
                                    self.all_player_names.remove(name_to_remove)
                                continue
                                
                            if 'msg' in msg:
                                raw_msg = msg['msg']
                                
                                if "MOONSHINE_BANNER|" in raw_msg:
                                    parts = raw_msg.split("|")[1].split(" vs ")
                                    self.moonshine_hunters = parts[0].split(",")
                                    self.moonshine_leaders = parts[1].split(",")
                                    self.moonshine_waiting = True
                                    self.show_moonshine_jug = False
                                    self.jug_shot = False
                                    self.player_cards.clear()
                                    self.bullseye_anim_card = None
                                    continue 
                                
                                if "QUICKDRAW_BANNER|" in raw_msg:
                                    self.quickdraw_tied_players = raw_msg.split("|")[1].split(",")
                                    self.quickdraw_waiting = True
                                    self.quickdraw_timer = 120  
                                    continue

                                self.log_game_text(raw_msg)
                                
                                if "GAME PAUSED" in raw_msg:
                                    self.is_paused = True
                                elif "GAME RESUMED" in raw_msg:
                                    self.is_paused = False
                                    
                                elif "DRAW YOUR WEAPONS!" in raw_msg:
                                    self.game_over = False
                                    self.overall_winner = ""
                                    self.is_quickdraw = False
                                    
                                elif "=== GAME OVER! The deck ran dry. ===" in raw_msg:
                                    self.player_cards.clear()
                                    self.bullseye_anim_card = None
                                    self.cards_dealt = False
                                    self.is_quickdraw = False
                                    
                                elif "is shufflin'" in raw_msg:
                                    if "SPECIAL QUICKDRAW" in raw_msg:
                                        self.is_quickdraw = True
                                    else:
                                        self.is_quickdraw = False
                                        
                                    self.bullseye_anim_card = None
                                    self.show_moonshine_jug = False
                                    for p_name, card in self.player_cards.items():
                                        card.revealed = False
                                        card.bullet_holes.clear()
                                        card.is_winner = False 
                                        card.has_tombstone = False
                                        card.x = self.res[0] + 200 
                                        card.target_x = self.res[0] + 200
                                        card.y = self.res[1] + 200

Response 2:
Ah, I see exactly what is happening. You discovered a classic network programming "race condition"!

In the previous version, the client was sending its introductory info to the server *before* it actually started its background thread to listen for messages. Because the server processes connections in milliseconds, it was instantly broadcasting the updated scoreboard back to the client, but the client's "ears" weren't turned on yet, causing it to completely miss that initial packet containing the AI players!

I have fixed this by reversing the order: the client now turns on its listening thread *before* it knocks on the server's door. I also added a safety flag (`self.ai_added = True`) directly in the server's initialization so it doesn't accidentally try to spawn duplicate bots later.

AI_NAMES_POOL = ['Mecha-Wyatt', 'Cyber-Doc', 'Holo-Jesse', 'Robo-Calamity', 'Synth-Billy', 'Auto-Annie', 'Bot-Cassidy', 'Gear-Wayne']

def get_local_ip():
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        s.connect(("8.8.8.8", 80))
        ip = s.getsockname()[0]
        s.close()
        return ip
    except Exception:
        return "127.0.0.1"

def create_deck():
    return [{'suit': s, 'value': v, 'rank': r} for s in SUITS for v, r in VALUES.items()]

def calculate_thumper(age, height, weight_kg):
    return round(age + height + weight_kg, 1)

def lerp(a, b, t):
    return a + (b - a) * t

def resource_path(relative_path):
    """ Get absolute path to resource, works for dev and for PyInstaller """
    try:
        # PyInstaller creates a temp folder and stores path in _MEIPASS
        base_path = sys._MEIPASS
    except Exception:
        base_path = os.path.abspath(".")

    return os.path.join(base_path, relative_path)

# --- Network Server ---
class TriggeredServer:
    def __init__(self, port, ai_count, num_decks=1):
        self.port = port
        self.ai_count = ai_count
        self.num_decks = num_decks
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1) 
        self.clients = {}  
        self.players = []  
        self.lock = threading.RLock() 
        self.trigger_calls = [] 
        self.game_active = False
        self.ai_added = True # Set to true so start_game_logic doesn't duplicate them
        self.round_num = 1
        self.trigger_phase = False 
        self.is_paused = False
        
        self.quickdraw_active = False
        self.quickdraw_players = []
        
        self.moonshine_active = False
        self.moonshine_risk_players = []
        self.moonshine_zero_players = []
        
        # Instantly spawn AIs into the lobby so they are on the board before the game starts
        available_ai_names = random.sample(AI_NAMES_POOL, min(self.ai_count, len(AI_NAMES_POOL)))
        for i in range(self.ai_count):
            ai_player = {
                'name': available_ai_names[i] if i < len(available_ai_names) else f"Bot-{i}",
                'thumper': round(random.uniform(90.0, 220.0), 1), 
                'is_ai': True,
                'score': 0
            }
            self.players.append(ai_player)
        
    def start(self):
        self.server_socket.bind(('0.0.0.0', self.port))
        self.server_socket.listen(8)
        threading.Thread(target=self.accept_clients, daemon=True).start()
        threading.Thread(target=self.timeout_checker, daemon=True).start()

    def timeout_checker(self):
        """ Runs in background to kick clients that haven't pinged in 30 seconds """
        while True:
            time.sleep(5)
            with self.lock:
                now = time.time()
                for conn, p_info in list(self.clients.items()):
                    # Give players 30 seconds before assuming they dropped
                    if now - p_info.get('last_seen', now) > 30:
                        self.remove_player(p_info['name'], "timed out")

    def remove_player(self, player_name, reason="disconnected"):
        """ Safely handles player removal, socket closure, and card redistribution """
        player_to_remove = next((p for p in self.players if p['name'] == player_name), None)
        if not player_to_remove: return
        
        # Close connection if it exists
        conn_to_close = next((conn for conn, p in self.clients.items() if p['name'] == player_name), None)
        if conn_to_close:
            try: conn_to_close.close()
            except: pass
            if conn_to_close in self.clients:
                del self.clients[conn_to_close]
        
        self.players.remove(player_to_remove)
        
        cards = player_to_remove.get('score', 0)
        self.broadcast({'msg': f"\n[!] {player_name} {reason} and left the table!"})
        
        if cards > 0 and self.players and self.game_active:
            min_score = min(p['score'] for p in self.players)
            lowest_players = [p for p in self.players if p['score'] == min_score]
            share = cards // len(lowest_players)
            remainder = cards % len(lowest_players)
            
            for i, p in enumerate(lowest_players):
                p['score'] += share + (1 if i < remainder else 0)
            
            lp_names = ", ".join([p['name'] for p in lowest_players])
            self.broadcast({'msg': f"Their {cards} cards were scattered to the lowest bounty hunters: {lp_names}."})
        
        # Tell all clients to visually delete this player's card from the screen
        self.broadcast({'action': 'remove_player', 'name': player_name})
        self.broadcast_scores()

    def accept_clients(self):
        while True:
            conn, addr = self.server_socket.accept()
            threading.Thread(target=self.handle_client, args=(conn,), daemon=True).start()

    def safe_sleep(self, duration):
        waited = 0
        while waited < duration:
            time.sleep(0.1)
            if not self.is_paused:
                waited += 0.1

    def register_trigger(self, player_name):
        with self.lock:
            # Block spectators from participating in quickdraws
            if self.game_active and self.quickdraw_active and player_name not in self.quickdraw_players:
                if not any(name == player_name for t, name in self.trigger_calls):
                    self.trigger_calls.append((time.time(), player_name))
                    self.broadcast({'msg': f"  [-] Calm down, {player_name}! It's not your turn to shoot."})
                return

            if self.trigger_phase:
                if not any(name == player_name for t, name in self.trigger_calls):
                    self.trigger_calls.append((time.time(), player_name))
                    if self.moonshine_active:
                        self.broadcast({'msg': f"  [!] *BANG* {player_name} shot at the jug!"})
                        player = next((p for p in self.players if p['name'] == player_name), None)
                        if player and player not in self.moonshine_risk_players and player not in self.moonshine_zero_players:
                            if player['score'] > 0:
                                self.moonshine_risk_players.append(player)
                                self.broadcast({'msg': f"  [!] {player_name} joins the fun and risks their {player['score']} cards!"})
                    else:
                        self.broadcast({'msg': f"  [!] *BANG* {player_name} reached for their iron!"})
            elif self.game_active:
                if not any(name == player_name for t, name in self.trigger_calls):
                    self.trigger_calls.append((time.time(), player_name))
                    self.broadcast({'msg': f"  [-] *Click* {player_name} drew too early! (Wait for the cue!)"})

    def handle_client(self, conn):
        try:
            data = conn.recv(1024).decode()
            if data:
                player_info = json.loads(data)
                player_info['conn'] = conn
                player_info['is_ai'] = False
                player_info['score'] = 0
                player_info['last_seen'] = time.time()
                
                with self.lock:
                    # Enforce Maximum 8 players
                    if len(self.players) >= 8:
                        ai_players = [p for p in self.players if p.get('is_ai')]
                        if ai_players:
                            # Boot an AI to make room for the human
                            ai_to_kick = random.choice(ai_players)
                            self.remove_player(ai_to_kick['name'], "was booted from the saloon to make room")
                        else:
                            # Table is full of actual humans
                            conn.sendall((json.dumps({'action': 'quit', 'msg': "The table is full! (Max 8 players)"}) + "\n").encode())
                            conn.close()
                            return # Exit thread

                    self.clients[conn] = player_info
                    self.players.append(player_info)
                    
                self.broadcast({'msg': f"Gunslinger {player_info['name']} moseyed into the saloon!"})
                self.broadcast_scores()
                
            buffer = ""
            while True:
                data = conn.recv(1024).decode()
                if not data: break
                buffer += data
                
                while "\n" in buffer:
                    line, buffer = buffer.split("\n", 1)
                    line = line.strip()
                    if not line: continue
                    
                    try:
                        parsed = json.loads(line)
                        action = parsed.get('action')
                        
                        if action == 'ping':
                            with self.lock:
                                if conn in self.clients:
                                    self.clients[conn]['last_seen'] = time.time()
                            continue
                        elif action == 'trigger' and self.game_active and not self.is_paused:
                            self.register_trigger(player_info['name'])
                        elif action == 'toggle_pause':
                            self.is_paused = not self.is_paused
                            status = "PAUSED" if self.is_paused else "RESUMED"
                            self.broadcast({'msg': f"\n--- GAME {status} ---"})
                        elif action == 'start' and not self.game_active:
                            threading.Thread(target=self.start_game_logic, daemon=True).start()
                        elif action == 'restart' and not self.game_active:
                            threading.Thread(target=self.start_game_logic, daemon=True).start()
                        elif action == 'exit':
                            self.broadcast({'action': 'quit', 'msg': "\nServer is packin' up. Happy trails, partners!"})
                            time.sleep(1)
                            os._exit(0) 
                    except Exception:
                        pass 
        except:
            pass
        finally:
            with self.lock:
                if conn in self.clients:
                    name = self.clients[conn]['name']
                    self.remove_player(name, "lost connection")

    def broadcast(self, message):
        msg_str = json.dumps(message) + "\n"
        with self.lock:
            for conn in list(self.clients.keys()):
                try:
                    conn.sendall(msg_str.encode())
                    time.sleep(0.01) 
                except:
                    pass

    def broadcast_scores(self):
        scores = {p['name']: p['score'] for p in self.players}
        self.broadcast({'action': 'update_scores', 'scores': scores})

    def start_game_logic(self):
        with self.lock:
            if self.game_active: return
            
            if not self.ai_added:
                available_ai_names = random.sample(AI_NAMES_POOL, min(self.ai_count, len(AI_NAMES_POOL)))
                spots_left = 8 - len(self.players)
                actual_ai_count = min(self.ai_count, max(0, spots_left))
                
                for i in range(actual_ai_count):
                    ai_player = {
                        'name': available_ai_names[i] if i < len(available_ai_names) else f"Bot-{i}",
                        'thumper': round(random.uniform(90.0, 220.0), 1), 
                        'is_ai': True,
                        'score': 0
                    }
                    self.players.append(ai_player)
                self.ai_added = True

            if len(self.players) < 2:
                self.broadcast({'msg': "Hold yer horses! We need at least 2 gunslingers to start."})
                return
                
            self.game_active = True
            self.round_num = 1
            for p in self.players:
                p['score'] = 0

        self.broadcast({'msg': "\n" + "="*40 + "\nDRAW YOUR WEAPONS! THE GAME BEGINS!\n" + "="*40})
        
        thumper_display = "\n--- THUMPER POWERS ---\n"
        for p in self.players:
            thumper_display += f"  {p['name']}: {p['thumper']}\n"
        thumper_display += "----------------------"
        self.broadcast({'msg': thumper_display})
        self.safe_sleep(4)

        self.broadcast({'msg': "\nJudgin' the First Dealer based on Thumper Power..."})
        max_thumper = max(p['thumper'] for p in self.players)
        tied_players = [p for p in self.players if p['thumper'] == max_thumper]
        
        if len(tied_players) > 1:
            self.broadcast({'msg': "We got ourselves a Mexican standoff! Resolvin' the tie..."})
            self.safe_sleep(2)
            first_dealer = random.choice(tied_players)
        else:
            first_dealer = tied_players[0]
            
        self.broadcast({'msg': f"The First Dealer is {first_dealer['name']}!"})
        self.safe_sleep(2)
        
        self.broadcast_scores()
        self.game_loop(first_dealer)

    def get_high_card_winners(self, round_cards):
        max_rank = max(c['rank'] for c in round_cards.values())
        return [name for name, c in round_cards.items() if c['rank'] == max_rank]

    def resolve_by_thumper(self, tied_objs, dealer):
        max_thumper = max(p['thumper'] for p in tied_objs)
        thump_winners = [p for p in tied_objs if p['thumper'] == max_thumper]
        if len(thump_winners) > 1:
            self.broadcast({'msg': f"Thumper power is tied! Dealer {dealer['name']} wins ties. House rules."})
            return dealer, 0
        else:
            winner = thump_winners[0]
            self.broadcast({'msg': f"{winner['name']} wins the tiebreaker with superior Thumper Power ({winner['thumper']})!"})
            return winner, 0

    def play_quickdraw_round(self, tied_player_names, dealer, deck, current_pot):
        self.quickdraw_active = True
        self.quickdraw_players = tied_player_names.copy()
        
        try:
            tied_objs = [p for p in self.players if p['name'] in tied_player_names]
            tie_pot = len(tied_objs) + 1
            total_pot = current_pot + tie_pot
            
            if len(deck) < tie_pot + 1:
                self.broadcast({'msg': "Deck is too low for a tiebreaker! Resolving by Thumper..."})
                w, _ = self.resolve_by_thumper(tied_objs, dealer)
                return w, tie_pot

            self.broadcast({'msg': f"QUICKDRAW_BANNER|{','.join(tied_player_names)}"})
            self.broadcast({'msg': f"\n--- SPECIAL QUICKDRAW ROUND --- Dealer is shufflin' tiebreaker cards..."})
            self.safe_sleep(2.0)
            
            round_cards = {}
            for p in tied_objs:
                round_cards[p['name']] = deck.pop()
            bullseye = deck.pop()
            
            self.broadcast({'msg': f"Cards dealt face down in the dirt. The Tiebreaker Bullseye is:"})
            self.broadcast({'msg': f"\n      >>> {bullseye['value']} of {bullseye['suit']} <<<\n"})
            
            self.safe_sleep(2)
            
            with self.lock:
                self.trigger_calls.clear()
                self.trigger_phase = True
                
            self.broadcast({'msg': f"{dealer['name']} draws and yells: 'CLICK CLICK'! (Turn 'em over!)"})
            
            cards_display = "Tiebreaker Cards revealed:\n" + "\n".join([f"  - {name}: {c['value']} of {c['suit']}" for name, c in round_cards.items()])
            self.broadcast({'msg': cards_display})
            
            match_exists = any(c['value'] == bullseye['value'] for c in round_cards.values())

            if match_exists:
                for p in tied_objs:
                    if p.get('is_ai'):
                        if random.random() < 0.15:
                            reaction = random.uniform(0.15, 0.7)
                        else:
                            reaction = random.uniform(0.8, 4.8)
                        threading.Timer(reaction, self.register_trigger, args=(p['name'],)).start()

                self.safe_sleep(TRIGGER_WINDOW)
                
                with self.lock:
                    self.trigger_phase = False 
                    # Only accept triggers from the tied participants
                    valid_calls = [c for c in self.trigger_calls if c[1] in tied_player_names]
                    if valid_calls:
                        valid_calls.sort(key=lambda x: x[0])
                        fastest_time = valid_calls[0][0]
                        ties = [name for t, name in valid_calls if t - fastest_time < 0.1]
                        
                        if len(ties) > 1:
                            self.broadcast({'msg': f"Another tie! Resolving by Thumper Power..."})
                            w, _ = self.resolve_by_thumper([p for p in self.players if p['name'] in ties], dealer)
                            return w, tie_pot
                        else:
                            winner_name = ties[0]
                            self.broadcast({'msg': f"*** {winner_name} WON THE TIEBREAKER! ***"})
                            w = next(p for p in self.players if p['name'] == winner_name)
                            return w, tie_pot
                    else:
                        self.broadcast({'msg': "Nobody pulled the trigger! Checking high card..."})
                        high_players = self.get_high_card_winners(round_cards)
                        if len(high_players) > 1:
                            self.broadcast({'msg': f"Tie for the high card! Resolving by Thumper Power..."})
                            w, _ = self.resolve_by_thumper([p for p in tied_objs if p['name'] in high_players], dealer)
                            return w, tie_pot
                        else:
                            winner_name = high_players[0]
                            self.broadcast({'msg': f"{winner_name} holds the highest tiebreaker card!"})
                            w = next(p for p in tied_objs if p['name'] == winner_name)
                            return w, tie_pot
            else:
                for p in tied_objs:
                    if p.get('is_ai'):
                        if random.random() < 0.05: 
                            reaction = random.uniform(0.5, 2.5)
                            threading.Timer(reaction, self.register_trigger, args=(p['name'],)).start()

                self.safe_sleep(3.0) 
                with self.lock:
                    self.trigger_phase = False
                    valid_calls = [c for c in self.trigger_calls if c[1] in tied_player_names]
                    if valid_calls:
                        valid_calls.sort(key=lambda x: x[0])
                        first_trigger_name = valid_calls[0][1]
                        
                        if first_trigger_name == dealer['name']:
                            self.broadcast({'msg': f"Misfire! The Dealer ({dealer['name']}) drew on a ghost!"})
                            self.broadcast({'msg': f"The dealer loses! {total_pot} cards are scattered randomly to the other players."})
                            other_players = [p for p in self.players if p['name'] != dealer['name']]
                            if other_players:
                                for _ in range(total_pot):
                                    random.choice(other_players)['score'] += 1
                            return None, -current_pot
                        else:
                            self.broadcast({'msg': f"Misfire! {first_trigger_name} got trigger-happy with no match."})
                            self.broadcast({'msg': f"Dealer {dealer['name']} takes the pot."})
                            return dealer, tie_pot
                    else:
                        high_players = self.get_high_card_winners(round_cards)
                        if len(high_players) > 1:
                            self.broadcast({'msg': f"Tie for the high card! Resolving by Thumper Power..."})
                            w, _ = self.resolve_by_thumper([p for p in tied_objs if p['name'] in high_players], dealer)
                            return w, tie_pot
                        else:
                            winner_name = high_players[0]
                            self.broadcast({'msg': f"{winner_name} holds the highest tiebreaker card!"})
                            w = next(p for p in tied_objs if p['name'] == winner_name)
                            return w, tie_pot
        finally:
            self.quickdraw_active = False
            self.quickdraw_players.clear()

    def moonshine_shootout(self, high_players, zero_players):
        self.moonshine_active = True
        self.moonshine_risk_players = high_players.copy()
        self.moonshine_zero_players = zero_players.copy()
        
        self.broadcast({'msg': "\n" + "="*40})
        self.broadcast({'msg': "!!! MOONSHINE SHOOTOUT BONUS ROUND !!!"})
        self.broadcast({'msg': "Zero-point cowboys get one last chance to steal the leader's bounty!"})
        
        risk_names = [p['name'] for p in high_players]
        zero_names = [p['name'] for p in zero_players]
        
        self.broadcast({'msg': f"Leaders (At Risk): {', '.join(risk_names)}"})
        self.broadcast({'msg': f"Zero-Pointers (Hunters): {', '.join(zero_names)}"})
        
        self.broadcast({'msg': f"MOONSHINE_BANNER|{','.join(zero_names)} vs {','.join(risk_names)}"})
        self.broadcast({'msg': "Leaders and Zero-pointers: GET READY TO SHOOT THE JUG!"})
        
        self.safe_sleep(random.uniform(3.5, 6.0))
        
        with self.lock:
            self.trigger_calls.clear()
            self.trigger_phase = True
            
        self.broadcast({'msg': "\n >>> *JUG OF MOONSHINE THROWN* CLICK CLICK! (Shoot now!) <<< \n"})
        
        participants = high_players + zero_players
        for p in participants:
            if p.get('is_ai'):
                if random.random() < 0.15: 
                    reaction = random.uniform(0.15, 0.6)
                else:
                    reaction = random.uniform(0.7, 2.4)
                threading.Timer(reaction, self.register_trigger, args=(p['name'],)).start()
                
        self.safe_sleep(JUG_AIR_TIME)
        
        with self.lock:
            self.trigger_phase = False
            self.moonshine_active = False 
            
            zero_names = [p['name'] for p in self.moonshine_zero_players]
            risk_names = [p['name'] for p in self.moonshine_risk_players]
            original_leaders = [p['name'] for p in high_players]
            
            zero_calls = [c for c in self.trigger_calls if c[1] in zero_names]
            risk_calls = [c for c in self.trigger_calls if c[1] in risk_names]
            
            # Find the fastest original leader to see if any voluntary joiners beat them
            leader_calls = [c for c in risk_calls if c[1] in original_leaders]
            leader_calls.sort(key=lambda x: x[0])
            fastest_leader_time = leader_calls[0][0] if leader_calls else float('inf')
            
            if not zero_calls:
                self.broadcast({'msg': "No zero-point player hit the jug! The leaders keep their stash."})
                self.broadcast_scores()
                return
                
            zero_calls.sort(key=lambda x: x[0])
            fastest_zero_time, fastest_zero_name = zero_calls[0]
            
            self.broadcast({'msg': f"*** {fastest_zero_name} shot the jug! ***"})
            total_stolen = 0
            stealer_pool = self.moonshine_zero_players.copy()
            
            for h in self.moonshine_risk_players:
                h_name = h['name']
                h_call = next((c for c in risk_calls if c[1] == h_name), None)
                is_voluntary = h_name not in original_leaders
                
                if not h_call or h_call[0] > fastest_zero_time:
                    stolen = h['score']
                    h['score'] = 0
                    total_stolen += stolen
                    self.broadcast({'msg': f"  -> {h_name} was too slow! They lost {stolen} cards!"})
                else:
                    self.broadcast({'msg': f"  -> {h_name} fired fast and defended their stash!"})
                    
                    # Reward voluntary risk players if they outperformed ALL leaders
                    if is_voluntary and h_call[0] < fastest_leader_time:
                        stealer_pool.append(h)
                        self.broadcast({'msg': f"  -> {h_name} outdrew the leaders and joins the hunters' split!"})
                        
            if total_stolen > 0:
                share = total_stolen // len(stealer_pool)
                remainder = total_stolen % len(stealer_pool)
                for i, p in enumerate(stealer_pool):
                    p['score'] += share + (1 if i < remainder else 0)
                
                pool_names = ", ".join([p['name'] for p in stealer_pool])
                self.broadcast({'msg': f"The stolen {total_stolen} cards are divided among: {pool_names}!"})
            else:
                self.broadcast({'msg': "The leaders survived the ambush!"})
                
            self.broadcast_scores()

    def game_loop(self, first_dealer):
        deck = []
        for _ in range(self.num_decks):
            deck.extend(create_deck())
        random.shuffle(deck)
        
        dealer_idx = self.players.index(first_dealer)
        original_dealer = first_dealer['name']
        
        # Adding check len(self.players) > 1 ensures game doesn't crash if everyone drops
        while len(deck) >= len(self.players) + 1 and len(self.players) > 1:
            dealer_idx = dealer_idx % len(self.players)
            dealer = self.players[dealer_idx]
            self.broadcast({'msg': f"\n--- ROUND {self.round_num} --- Dealer {dealer['name']} is shufflin'..."})
            self.safe_sleep(1.5)
            
            round_cards = {}
            for p in self.players:
                round_cards[p['name']] = deck.pop()
            bullseye = deck.pop()
            
            self.broadcast({'msg': f"Cards dealt face down in the dirt. The Bullseye is:"})
            self.broadcast({'msg': f"\n      >>> {bullseye['value']} of {bullseye['suit']} <<<\n"})
            
            self.safe_sleep(2)
            
            with self.lock:
                self.trigger_calls.clear()
                self.trigger_phase = True
                
            self.broadcast({'msg': f"{dealer['name']} draws and yells: 'CLICK CLICK'! (Turn 'em over!)"})
            
            cards_display = "Cards revealed on the table:\n" + "\n".join([f"  - {name}: {c['value']} of {c['suit']}" for name, c in round_cards.items()])
            self.broadcast({'msg': cards_display})
            
            match_exists = any(c['value'] == bullseye['value'] for c in round_cards.values())
            pot = len(self.players) + 1
            winner = None

            if match_exists:
                for p in self.players:
                    if p.get('is_ai'):
                        if random.random() < 0.15: 
                            reaction = random.uniform(0.15, 0.7)
                        else:
                            reaction = random.uniform(0.8, 4.8)
                        threading.Timer(reaction, self.register_trigger, args=(p['name'],)).start()

                self.safe_sleep(TRIGGER_WINDOW)
                
                initiate_quickdraw = False
                qd_ties = []
                
                with self.lock:
                    self.trigger_phase = False 
                    
                    if self.trigger_calls:
                        self.trigger_calls.sort(key=lambda x: x[0])
                        fastest_time = self.trigger_calls[0][0]
                        ties = [name for t, name in self.trigger_calls if t - fastest_time < 0.1]
                        
                        if len(ties) > 1:
                            initiate_quickdraw = True
                            qd_ties = ties
                        else:
                            winner_name = ties[0]
                            self.broadcast({'msg': f"*** {winner_name} WAS THE FASTEST GUN! ***"})
                            winner = next(p for p in self.players if p['name'] == winner_name)
                    else:
                        self.broadcast({'msg': "Nobody pulled the trigger! Let's see who's holdin' the high card..."})
                        high_players = self.get_high_card_winners(round_cards)
                        if len(high_players) > 1:
                            initiate_quickdraw = True
                            qd_ties = high_players
                        else:
                            winner_name = high_players[0]
                            self.broadcast({'msg': f"{winner_name} holds the highest card!"})
                            winner = next(p for p in self.players if p['name'] == winner_name)

                if initiate_quickdraw:
                    self.broadcast({'msg': f"Trigger TIE between {', '.join(qd_ties)}! Initiating Quickdraw Round!"})
                    winner, extra_pot = self.play_quickdraw_round(qd_ties, dealer, deck, pot)
                    pot += extra_pot

            else:
                for p in self.players:
                    if p.get('is_ai'):
                        if random.random() < 0.02: 
                            reaction = random.uniform(0.8, 3.5)
                            threading.Timer(reaction, self.register_trigger, args=(p['name'],)).start()

                self.safe_sleep(3.0) 
                
                initiate_quickdraw = False
                qd_ties = []
                
                with self.lock:
                    self.trigger_phase = False
                    if self.trigger_calls:
                        self.trigger_calls.sort(key=lambda x: x[0])
                        first_trigger_name = self.trigger_calls[0][1]
                        
                        if first_trigger_name == dealer['name']:
                            self.broadcast({'msg': f"Misfire! The Dealer ({dealer['name']}) drew on a ghost!"})
                            self.broadcast({'msg': f"The dealer loses! {pot} cards are scattered randomly to the other players."})
                            other_players = [p for p in self.players if p['name'] != dealer['name']]
                            if other_players:
                                for _ in range(pot):
                                    random.choice(other_players)['score'] += 1
                            winner = None 
                        else:
                            self.broadcast({'msg': f"Misfire! {first_trigger_name} got trigger-happy with no match."})
                            self.broadcast({'msg': f"Dealer {dealer['name']} takes the pot."})
                            winner = dealer
                    else:
                        high_players = self.get_high_card_winners(round_cards)
                        if len(high_players) > 1:
                            initiate_quickdraw = True
                            qd_ties = high_players
                        else:
                            winner_name = high_players[0]
                            self.broadcast({'msg': f"{winner_name} holds the highest card!"})
                            winner = next((p for p in self.players if p['name'] == winner_name), None)

                if initiate_quickdraw:
                    self.broadcast({'msg': f"Tie for the high card between {', '.join(qd_ties)}! Initiating Quickdraw Round!"})
                    winner, extra_pot = self.play_quickdraw_round(qd_ties, dealer, deck, pot)
                    pot += extra_pot

            if winner:
                winner['score'] += pot
                self.broadcast({'msg': f"{winner['name']} wins the round and collects {pot} cards!\n"})
            else:
                self.broadcast({'msg': "The round ends in chaos! Cards distributed.\n"})
            
            self.broadcast_scores()
            self.round_num += 1
            dealer_idx = (dealer_idx + 1) % len(self.players)
            self.safe_sleep(2.5)

        self.broadcast({'msg': "\n=== GAME OVER! The deck ran dry. ==="})
        remaining = len(deck)
        
        # Failsafe if the original dealer was kicked out mid-game
        first_dealer_obj = next((p for p in self.players if p['name'] == original_dealer), None)
        if first_dealer_obj:
            first_dealer_obj['score'] += remaining
            self.broadcast({'msg': f"First dealer {original_dealer} pockets the remaining {remaining} cards."})
            
        self.broadcast_scores()
        self.safe_sleep(2.0)
        
        if self.players:
            # Check for Moonshine Shootout
            high_score = max(p['score'] for p in self.players)
            high_players = [p for p in self.players if p['score'] == high_score]
            zero_players = [p for p in self.players if p['score'] == 0]
            
            if high_score > 0 and len(zero_players) > 0:
                self.moonshine_shootout(high_players, zero_players)
                self.safe_sleep(3.0)
            
            self.players.sort(key=lambda x: x['score'], reverse=True)
            standings = "\nFINAL BOUNTIES (Scores):\n" + "\n".join([f"  {p['name']}: {p['score']} cards" for p in self.players])
            self.broadcast({'msg': standings})
            
        self.game_active = False
        
        # FINAL BOUNTY BOARD UPDATE
        self.broadcast_scores() 
        self.broadcast({'msg': "\nHost can press 'R' to play a new game, or close the window to exit."})

class FloatingMessage:
    def __init__(self, text, color, start_y):
        self.text = text
        self.color = color
        self.y = start_y
        self.target_y = start_y

class AnimatedCard:
    def __init__(self, start_x, start_y, value=None, suit=None, label=""):
        self.x = start_x
        self.y = start_y
        self.target_x = start_x
        self.target_y = start_y
        self.draw_x = start_x
        self.draw_y = start_y
        
        self.value = value
        self.suit = suit
        self.label = label
        self.revealed = False
        
        self.bullet_holes = []  
        self.has_tombstone = False
        self.is_winner = False
        self.shake_timer = 0
        self.flash_alpha = 0

    def update(self):
        self.x = lerp(self.x, self.target_x, 0.1)
        self.y = lerp(self.y, self.target_y, 0.1)
        
        self.draw_x = self.x
        self.draw_y = self.y
        
        if self.shake_timer > 0:
            self.draw_x += random.randint(-5, 5)
            self.draw_y += random.randint(-5, 5)
            self.shake_timer -= 1
            
        if self.flash_alpha > 0:
            self.flash_alpha = max(0, self.flash_alpha - 8)

    def evaluate_tumbleweeds(self):
        out_of_bounds = False

            if self.setup_data['AGE']:

        except ValueError: pass
        
        try:
            if self.setup_data['HEIGHT']:
                v = float(self.setup_data['HEIGHT'])
                if v < 4.5 or v > 6.5: out_of_bounds = True
        except ValueError: pass
        
        try:
            if self.setup_data['WEIGHT']:
                v = float(self.setup_data['WEIGHT'])
                if v < 100 or v > 260: out_of_bounds = True
        except ValueError: pass
        
        if out_of_bounds:
            if self.tumbleweed_state in ['OFF', 'BLOWING_AWAY']:
                if self.tumbleweed_state == 'OFF':
                    self.reset_tumbleweeds()
                self.tumbleweed_state = 'ROLLING'
        else:
            if self.tumbleweed_state == 'ROLLING':
                self.tumbleweed_state = 'BLOWING_AWAY'

    def validate_field(self, step_id, val):
        try:
            if step_id == 'MODE':
                if val.lower() not in ['s', 'c']: return False, "Type 'S' for Host or 'C' for Client."
                self.is_host = (val.lower() == 's')
                
                if self.is_host:
                    self.setup_sequence = ['MODE', 'NAME', 'AGE', 'HEIGHT', 'WEIGHT', 'DECKS', 'NETWORK', 'PORT']
                else:
                    self.setup_sequence = ['MODE', 'NAME', 'AGE', 'HEIGHT', 'WEIGHT', 'NETWORK', 'PORT']
                    
            elif step_id == 'NAME':
                if not val.strip(): return False, "Name cannot be empty."
                self.name = val.strip()
            elif step_id == 'AGE':
                v = float(val)
                if v <= 0: return False, "Age must be > 0."
                self.age = v
            elif step_id == 'HEIGHT':
                v = float(val)
                if v <= 0: return False, "Height must be > 0."
                self.height = v
            elif step_id == 'WEIGHT':
                v = float(val)
                if v <= 0: return False, "Weight must be > 0."
                self.weight_lbs = v
                self.thumper = calculate_thumper(self.age, self.height, self.weight_lbs * 0.453592)
            elif step_id == 'DECKS':
                v = int(val)
                if not (1 <= v <= 5): return False, "Decks must be 1-5."
                self.num_decks = v
            elif step_id == 'NETWORK':
                if self.is_host:
                    v = int(val)
                    if not (0 <= v <= 7): return False, "AI count 0-7."
                    self.ai_count = v
                else:
                    if not val.strip(): return False, "IP cannot be empty."
                    self.host_ip = val.strip()
            elif step_id == 'PORT':
                v = int(val)
                if not (1024 < v <= 65535): return False, "Port 1025-65535."
                self.port = v
            return True, ""
        except ValueError:
            return False, "Invalid format. Numbers required here."

    def handle_setup_input(self, event):
        current_field = self.setup_sequence[self.setup_idx]
        
        if event.key == pygame.K_BACKSPACE:
            self.setup_data[current_field] = self.setup_data[current_field][:-1]
            self.setup_error = ""
        elif event.key == pygame.K_UP:
            if self.setup_idx > 0:
                self.setup_idx -= 1
                self.setup_error = ""
                self.evaluate_tumbleweeds()
        elif event.key == pygame.K_DOWN:
            valid, err = self.validate_field(current_field, self.setup_data[current_field])
            if valid:
                if self.setup_idx < len(self.setup_sequence) - 1:
                    self.setup_idx += 1
                    self.setup_error = ""
                    self.evaluate_tumbleweeds()
            else:
                self.setup_error = err
        elif event.key in (pygame.K_RETURN, pygame.K_KP_ENTER):
            valid, err = self.validate_field(current_field, self.setup_data[current_field])
            if valid:
                if self.setup_idx < len(self.setup_sequence) - 1:
                    self.setup_idx += 1
                    self.setup_error = ""
                    self.evaluate_tumbleweeds()
                else:
                    all_valid = True
                    for i, f in enumerate(self.setup_sequence):
                        v, e = self.validate_field(f, self.setup_data[f])
                        if not v:
                            self.setup_idx = i
                            self.setup_error = e
                            all_valid = False
                            break
                    if all_valid:
                        self.finalize_setup()
            else:
                self.setup_error = err
        else:
            if event.unicode.isprintable() and len(self.setup_data[current_field]) < 20:
                self.setup_data[current_field] += event.unicode
                self.setup_error = ""

    def finalize_setup(self):
        pygame.display.set_caption("TRIGGERED - High Res Edition" + (" [HOST]" if self.is_host else ""))
        
        if self.is_host:
            # Bypass the loop validator to guarantee the AI count variable is rigidly populated
            try:
                ai_qty = int(self.setup_data.get('NETWORK', '0'))
            except:
                ai_qty = 0
            self.ai_count = ai_qty
            
            self.server = TriggeredServer(self.port, self.ai_count, self.num_decks)
            self.server.start()
            time.sleep(0.5)
            self.host_ip = '127.0.0.1'
            
        self.connect_to_server()

    def send_heartbeats(self):
        """ Keeps the client connection alive by pinging the server """
        while self.running and self.client_socket:
            try:
                msg = json.dumps({'action': 'ping'}) + "\n"
                self.client_socket.sendall(msg.encode())
            except:
                pass
            time.sleep(5)

    def connect_to_server(self):
        self.client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        try:
            self.client_socket.connect((self.host_ip, self.port))
            
            self.state = 'TITLE'
            
            # Start threads first to prevent missing the initial score broadcast
            threading.Thread(target=self.receive_messages, daemon=True).start()
            threading.Thread(target=self.send_heartbeats, daemon=True).start()
            
            info = json.dumps({'name': self.name, 'thumper': self.thumper})
            self.client_socket.sendall(info.encode())
            
            self.add_log_message("Welcome to the Saloon!", SVGA['TEXT_LIGHT'])
            self.log_game_text("Welcome to the Saloon!")
            
            if self.is_host:
                self.add_log_message("You are the Host. Press 'S' to Start.", SVGA['GOLD'])
                self.log_game_text("You are the Host. Press 'S' to Start.")
            else:
                self.add_log_message("Waiting for the Host to start...", SVGA['TEXT_LIGHT'])
                self.log_game_text("Waiting for the Host to start...")
                
            self.add_log_message("Press 'T' to pull the TRIGGER!", SVGA['TEXT_LIGHT'])
            self.log_game_text("Press 'T' to pull the TRIGGER!")
            self.add_log_message("Press 'Q' to toggle audio.", SVGA['TEXT_LIGHT'])
            self.log_game_text("Press 'Q' to toggle audio.")
            self.add_log_message("Press 'P' to Pause.", SVGA['TEXT_LIGHT'])
            self.log_game_text("Press 'P' to Pause.")
            
        except ConnectionRefusedError:
            self.setup_idx = self.setup_sequence.index('PORT')
            self.setup_error = "Connection Failed. Check IP/Port and try again."

    def receive_messages(self):
        buffer = ""
        while self.running and self.state in ['TITLE', 'DOOR_ANIM', 'GAME']:
            try:
                data = self.client_socket.recv(2048).decode()
                if not data: break
                buffer += data
                
                while "\n" in buffer:
                    line, buffer = buffer.split("\n", 1)
                    line = line.strip()
                    if not line: continue
                    
                    try:
                        msg = json.loads(line)
                        
                        if isinstance(msg, dict):
                            if msg.get('action') == 'update_scores':
                                self.current_scores = msg.get('scores', {})
                                continue
                                
                            if msg.get('action') == 'remove_player':
                                name_to_remove = msg.get('name')
                                if name_to_remove in self.player_cards:
                                    del self.player_cards[name_to_remove]
                                if name_to_remove in self.all_player_names:
                                    self.all_player_names.remove(name_to_remove)
                                continue
                                
                            if 'msg' in msg:
                                raw_msg = msg['msg']
                                
                                if "MOONSHINE_BANNER|" in raw_msg:
                                    parts = raw_msg.split("|")[1].split(" vs ")
                                    self.moonshine_hunters = parts[0].split(",")
                                    self.moonshine_leaders = parts[1].split(",")
                                    self.moonshine_waiting = True
                                    self.show_moonshine_jug = False
                                    self.jug_shot = False
                                    self.player_cards.clear()
                                    self.bullseye_anim_card = None
                                    continue 
                                
                                if "QUICKDRAW_BANNER|" in raw_msg:
                                    self.quickdraw_tied_players = raw_msg.split("|")[1].split(",")
                                    self.quickdraw_waiting = True
                                    self.quickdraw_timer = 120  
                                    continue

                                self.log_game_text(raw_msg)
                                
                                if "GAME PAUSED" in raw_msg:
                                    self.is_paused = True
                                elif "GAME RESUMED" in raw_msg:
                                    self.is_paused = False
                                    
                                elif "DRAW YOUR WEAPONS!" in raw_msg:
                                    self.game_over = False
                                    self.overall_winner = ""
                                    self.is_quickdraw = False
                                    
                                elif "=== GAME OVER! The deck ran dry. ===" in raw_msg:
                                    self.player_cards.clear()
                                    self.bullseye_anim_card = None
                                    self.cards_dealt = False
                                    self.is_quickdraw = False
                                    
                                elif "is shufflin'" in raw_msg:
                                    if "SPECIAL QUICKDRAW" in raw_msg:
                                        self.is_quickdraw = True
                                    else:
                                        self.is_quickdraw = False
                                        
                                    self.bullseye_anim_card = None
                                    self.show_moonshine_jug = False
                                    for p_name, card in self.player_cards.items():
                                        card.revealed = False
                                        card.bullet_holes.clear()
                                        card.is_winner = False 
                                        card.has_tombstone = False
                                        card.x = self.res[0] + 200 
                                        card.target_x = self.res[0] + 200
                                        card.y = self.res[1] + 200
                                        card.target_y = self.res[1] + 200
                                    self.cards_dealt = False
                                    
                                elif "--- THUMPER POWERS ---" in raw_msg:
                                    self.player_cards.clear()
                                    self.all_player_names = []
                                    idx = 0
                                    for m_line in raw_msg.split('\n'):
                                        if ':' in m_line and not '---' in m_line and not 'THUMPER' in m_line:

                                            if p_name:
                                                self.all_player_names.append(p_name)
                                                self.player_cards[p_name] = AnimatedCard(self.res[0] + 200, self.res[1] + 200, label=p_name)
                                                idx += 1

When the heartbeat feature was added, the client started silently sending a background "ping" signal to the server every 5 seconds. Because the connection happened so fast, the client was occasionally sending the initial setup data *and* the very first ping at the exact same millisecond.

The network combined them into a single blob of data (e.g., `{"name": "Corey"}{"action": "ping"}`). When the server tried to read that as a single JSON object, it crashed, instantly kicking you out behind the scenes. This is why the server ignored your 'S' and 'P' commands, and why it never sent the scoreboard data back to you!

I have completely rewritten the server's data receiver to cleanly separate and queue incoming network traffic line-by-line so it can never choke on stacked packets again.         return "127.0.0.1"

def create_deck():
    return [{'suit': s, 'value': v, 'rank': r} for s in SUITS for v, r in VALUES.items()]

def calculate_thumper(age, height, weight_kg):
    return round(age + height + weight_kg, 1)

def lerp(a, b, t):
    return a + (b - a) * t

def resource_path(relative_path):
    """ Get absolute path to resource, works for dev and for PyInstaller """
    try:
        # PyInstaller creates a temp folder and stores path in _MEIPASS
        base_path = sys._MEIPASS
    except Exception:
        base_path = os.path.abspath(".")

    return os.path.join(base_path, relative_path)

# --- Network Server ---
class TriggeredServer:
    def __init__(self, port, ai_count, num_decks=1):
        self.port = port
        self.ai_count = ai_count
        self.num_decks = num_decks
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1) 
        self.clients = {}  
        self.players = []  
        self.lock = threading.RLock() 
        self.trigger_calls = [] 
        self.game_active = False
        self.ai_added = True # Set to true so start_game_logic doesn't duplicate them
        self.round_num = 1
        self.trigger_phase = False 
        self.is_paused = False
        
        self.quickdraw_active = False
        self.quickdraw_players = []
        
        self.moonshine_active = False
        self.moonshine_risk_players = []
        self.moonshine_zero_players = []
        
        # Instantly spawn AIs into the lobby so they are on the board before the game starts
        available_ai_names = random.sample(AI_NAMES_POOL, min(self.ai_count, len(AI_NAMES_POOL)))
        for i in range(self.ai_count):
            ai_player = {
                'name': available_ai_names[i] if i < len(available_ai_names) else f"Bot-{i}",
                'thumper': round(random.uniform(90.0, 220.0), 1), 
                'is_ai': True,
                'score': 0
            }
            self.players.append(ai_player)
        
    def start(self):
        self.server_socket.bind(('0.0.0.0', self.port))
        self.server_socket.listen(8)
        threading.Thread(target=self.accept_clients, daemon=True).start()
        threading.Thread(target=self.timeout_checker, daemon=True).start()

    def timeout_checker(self):
        """ Runs in background to kick clients that haven't pinged in 30 seconds """
        while True:
            time.sleep(5)
            with self.lock:
                now = time.time()
                for conn, p_info in list(self.clients.items()):
                    # Give players 30 seconds before assuming they dropped
                    if now - p_info.get('last_seen', now) > 30:
                        self.remove_player(p_info['name'], "timed out")

    def remove_player(self, player_name, reason="disconnected"):
        """ Safely handles player removal, socket closure, and card redistribution """
        player_to_remove = next((p for p in self.players if p['name'] == player_name), None)
        if not player_to_remove: return
        
        # Close connection if it exists
        conn_to_close = next((conn for conn, p in self.clients.items() if p['name'] == player_name), None)
        if conn_to_close:
            try: conn_to_close.close()
            except: pass
            if conn_to_close in self.clients:
                del self.clients[conn_to_close]
        
        self.players.remove(player_to_remove)
        
        cards = player_to_remove.get('score', 0)
        self.broadcast({'msg': f"\n[!] {player_name} {reason} and left the table!"})
        
        if cards > 0 and self.players and self.game_active:
            min_score = min(p['score'] for p in self.players)
            lowest_players = [p for p in self.players if p['score'] == min_score]
            share = cards // len(lowest_players)
            remainder = cards % len(lowest_players)
            
            for i, p in enumerate(lowest_players):
                p['score'] += share + (1 if i < remainder else 0)
            
            lp_names = ", ".join([p['name'] for p in lowest_players])
            self.broadcast({'msg': f"Their {cards} cards were scattered to the lowest bounty hunters: {lp_names}."})
        
        # Tell all clients to visually delete this player's card from the screen
        self.broadcast({'action': 'remove_player', 'name': player_name})
        self.broadcast_scores()

    def accept_clients(self):
        while True:
            conn, addr = self.server_socket.accept()
            threading.Thread(target=self.handle_client, args=(conn,), daemon=True).start()

    def safe_sleep(self, duration):
        waited = 0
        while waited < duration:
            time.sleep(0.1)
            if not self.is_paused:
                waited += 0.1

    def register_trigger(self, player_name):
        with self.lock:
            # Block spectators from participating in quickdraws
            if self.game_active and self.quickdraw_active and player_name not in self.quickdraw_players:
                if not any(name == player_name for t, name in self.trigger_calls):
                    self.trigger_calls.append((time.time(), player_name))
                    self.broadcast({'msg': f"  [-] Calm down, {player_name}! It's not your turn to shoot."})
                return

            if self.trigger_phase:
                if not any(name == player_name for t, name in self.trigger_calls):
                    self.trigger_calls.append((time.time(), player_name))
                    if self.moonshine_active:
                        self.broadcast({'msg': f"  [!] *BANG* {player_name} shot at the jug!"})
                        player = next((p for p in self.players if p['name'] == player_name), None)
                        if player and player not in self.moonshine_risk_players and player not in self.moonshine_zero_players:
                            if player['score'] > 0:
                                self.moonshine_risk_players.append(player)
                                self.broadcast({'msg': f"  [!] {player_name} joins the fun and risks their {player['score']} cards!"})
                    else:
                        self.broadcast({'msg': f"  [!] *BANG* {player_name} reached for their iron!"})
            elif self.game_active:
                if not any(name == player_name for t, name in self.trigger_calls):
                    self.trigger_calls.append((time.time(), player_name))
                    self.broadcast({'msg': f"  [-] *Click* {player_name} drew too early! (Wait for the cue!)"})

    def handle_client(self, conn):
        player_info = None
        try:
            buffer = ""
            while True:
                data = conn.recv(1024).decode()
                if not data: break
                buffer += data
                
                # Process the buffer line by line to prevent JSON crashing on stacked TCP packets
                while "\n" in buffer:
                    line, buffer = buffer.split("\n", 1)
                    line = line.strip()
                    if not line: continue
                    
                    try:
                        parsed = json.loads(line)
                        
                        # The first packet received from a client will ALWAYS be their player info
                        if player_info is None:
                            player_info = parsed
                            player_info['conn'] = conn
                            player_info['is_ai'] = False
                            player_info['score'] = 0
                            player_info['last_seen'] = time.time()
                            
                            with self.lock:
                                # Enforce Maximum 8 players
                                if len(self.players) >= 8:
                                    ai_players = [p for p in self.players if p.get('is_ai')]
                                    if ai_players:
                                        # Boot an AI to make room for the human
                                        ai_to_kick = random.choice(ai_players)
                                        self.remove_player(ai_to_kick['name'], "was booted from the saloon to make room")
                                    else:
                                        # Table is full of actual humans
                                        conn.sendall((json.dumps({'action': 'quit', 'msg': "The table is full! (Max 8 players)"}) + "\n").encode())
                                        conn.close()
                                        return # Exit thread

                                self.clients[conn] = player_info
                                self.players.append(player_info)
                                
                            self.broadcast({'msg': f"Gunslinger {player_info['name']} moseyed into the saloon!"})
                            self.broadcast_scores()
                        else:
                            # Standard action processing
                            action = parsed.get('action')
                            
                            if action == 'ping':
                                with self.lock:
                                    if conn in self.clients:
                                        self.clients[conn]['last_seen'] = time.time()
                            elif action == 'trigger' and self.game_active and not self.is_paused:
                                self.register_trigger(player_info['name'])
                            elif action == 'toggle_pause':
                                self.is_paused = not self.is_paused
                                status = "PAUSED" if self.is_paused else "RESUMED"
                                self.broadcast({'msg': f"\n--- GAME {status} ---"})
                            elif action == 'start' and not self.game_active:
                                threading.Thread(target=self.start_game_logic, daemon=True).start()
                            elif action == 'restart' and not self.game_active:
                                threading.Thread(target=self.start_game_logic, daemon=True).start()
                            elif action == 'exit':
                                self.broadcast({'action': 'quit', 'msg': "\nServer is packin' up. Happy trails, partners!"})
                                time.sleep(1)
                                os._exit(0) 
                    except Exception:
                        pass # Ignore malformed packets
        except:
            pass
        finally:
            with self.lock:
                if player_info and conn in self.clients:
                    name = self.clients[conn]['name']
                    self.remove_player(name, "lost connection")

    def broadcast(self, message):
        msg_str = json.dumps(message) + "\n"
        with self.lock:
            for conn in list(self.clients.keys()):
                try:
                    conn.sendall(msg_str.encode())
                    time.sleep(0.01) 
                except:
                    pass

    def broadcast_scores(self):
        scores = {p['name']: p['score'] for p in self.players}
        self.broadcast({'action': 'update_scores', 'scores': scores})

    def start_game_logic(self):
        with self.lock:
            if self.game_active: return
            
            if len(self.players) < 2:
                self.broadcast({'msg': "Hold yer horses! We need at least 2 gunslingers to start."})
                return
                
            self.game_active = True
            self.round_num = 1
            for p in self.players:
                p['score'] = 0

        self.broadcast({'msg': "\n" + "="*40 + "\nDRAW YOUR WEAPONS! THE GAME BEGINS!\n" + "="*40})
        
        thumper_display = "\n--- THUMPER POWERS ---\n"
        for p in self.players:
            thumper_display += f"  {p['name']}: {p['thumper']}\n"
        thumper_display += "----------------------"
        self.broadcast({'msg': thumper_display})
        self.safe_sleep(4)

        self.broadcast({'msg': "\nJudgin' the First Dealer based on Thumper Power..."})
        max_thumper = max(p['thumper'] for p in self.players)
        tied_players = [p for p in self.players if p['thumper'] == max_thumper]
        
        if len(tied_players) > 1:
            self.broadcast({'msg': "We got ourselves a Mexican standoff! Resolvin' the tie..."})
            self.safe_sleep(2)
            first_dealer = random.choice(tied_players)
        else:
            first_dealer = tied_players[0]
            
        self.broadcast({'msg': f"The First Dealer is {first_dealer['name']}!"})
        self.safe_sleep(2)
        
        self.broadcast_scores()
        self.game_loop(first_dealer)

    def get_high_card_winners(self, round_cards):
        max_rank = max(c['rank'] for c in round_cards.values())
        return [name for name, c in round_cards.items() if c['rank'] == max_rank]

    def resolve_by_thumper(self, tied_objs, dealer):
        max_thumper = max(p['thumper'] for p in tied_objs)
        thump_winners = [p for p in tied_objs if p['thumper'] == max_thumper]
        if len(thump_winners) > 1:
            self.broadcast({'msg': f"Thumper power is tied! Dealer {dealer['name']} wins ties. House rules."})
            return dealer, 0
        else:
            winner = thump_winners[0]
            self.broadcast({'msg': f"{winner['name']} wins the tiebreaker with superior Thumper Power ({winner['thumper']})!"})
            return winner, 0

    def play_quickdraw_round(self, tied_player_names, dealer, deck, current_pot):
        self.quickdraw_active = True
        self.quickdraw_players = tied_player_names.copy()
        
        try:
            tied_objs = [p for p in self.players if p['name'] in tied_player_names]
            tie_pot = len(tied_objs) + 1
            total_pot = current_pot + tie_pot
            
            if len(deck) < tie_pot + 1:
                self.broadcast({'msg': "Deck is too low for a tiebreaker! Resolving by Thumper..."})
                w, _ = self.resolve_by_thumper(tied_objs, dealer)
                return w, tie_pot

            self.broadcast({'msg': f"QUICKDRAW_BANNER|{','.join(tied_player_names)}"})
            self.broadcast({'msg': f"\n--- SPECIAL QUICKDRAW ROUND --- Dealer is shufflin' tiebreaker cards..."})
            self.safe_sleep(2.0)
            
            round_cards = {}
            for p in tied_objs:
                round_cards[p['name']] = deck.pop()
            bullseye = deck.pop()
            
            self.broadcast({'msg': f"Cards dealt face down in the dirt. The Tiebreaker Bullseye is:"})
            self.broadcast({'msg': f"\n      >>> {bullseye['value']} of {bullseye['suit']} <<<\n"})
            
            self.safe_sleep(2)
            
            with self.lock:
                self.trigger_calls.clear()
                self.trigger_phase = True
                
            self.broadcast({'msg': f"{dealer['name']} draws and yells: 'CLICK CLICK'! (Turn 'em over!)"})
            
            cards_display = "Tiebreaker Cards revealed:\n" + "\n".join([f"  - {name}: {c['value']} of {c['suit']}" for name, c in round_cards.items()])
            self.broadcast({'msg': cards_display})
            
            match_exists = any(c['value'] == bullseye['value'] for c in round_cards.values())

            if match_exists:
                for p in tied_objs:
                    if p.get('is_ai'):
                        if random.random() < 0.15:
                            reaction = random.uniform(0.15, 0.7)
                        else:
                            reaction = random.uniform(0.8, 4.8)
                        threading.Timer(reaction, self.register_trigger, args=(p['name'],)).start()

                self.safe_sleep(TRIGGER_WINDOW)
                
                with self.lock:
                    self.trigger_phase = False 
                    # Only accept triggers from the tied participants
                    valid_calls = [c for c in self.trigger_calls if c[1] in tied_player_names]
                    if valid_calls:
                        valid_calls.sort(key=lambda x: x[0])
                        fastest_time = valid_calls[0][0]
                        ties = [name for t, name in valid_calls if t - fastest_time < 0.1]
                        
                        if len(ties) > 1:
                            self.broadcast({'msg': f"Another tie! Resolving by Thumper Power..."})
                            w, _ = self.resolve_by_thumper([p for p in self.players if p['name'] in ties], dealer)
                            return w, tie_pot
                        else:
                            winner_name = ties[0]
                            self.broadcast({'msg': f"*** {winner_name} WON THE TIEBREAKER! ***"})
                            w = next(p for p in self.players if p['name'] == winner_name)
                            return w, tie_pot
                    else:
                        self.broadcast({'msg': "Nobody pulled the trigger! Checking high card..."})
                        high_players = self.get_high_card_winners(round_cards)
                        if len(high_players) > 1:
                            self.broadcast({'msg': f"Tie for the high card! Resolving by Thumper Power..."})
                            w, _ = self.resolve_by_thumper([p for p in tied_objs if p['name'] in high_players], dealer)
                            return w, tie_pot
                        else:
                            winner_name = high_players[0]
                            self.broadcast({'msg': f"{winner_name} holds the highest tiebreaker card!"})
                            w = next(p for p in tied_objs if p['name'] == winner_name)
                            return w, tie_pot
            else:
                for p in tied_objs:
                    if p.get('is_ai'):
                        if random.random() < 0.05: 
                            reaction = random.uniform(0.5, 2.5)
                            threading.Timer(reaction, self.register_trigger, args=(p['name'],)).start()

                self.safe_sleep(3.0) 
                with self.lock:
                    self.trigger_phase = False
                    valid_calls = [c for c in self.trigger_calls if c[1] in tied_player_names]
                    if valid_calls:
                        valid_calls.sort(key=lambda x: x[0])
                        first_trigger_name = valid_calls[0][1]
                        
                        if first_trigger_name == dealer['name']:
                            self.broadcast({'msg': f"Misfire! The Dealer ({dealer['name']}) drew on a ghost!"})
                            self.broadcast({'msg': f"The dealer loses! {total_pot} cards are scattered randomly to the other players."})
                            other_players = [p for p in self.players if p['name'] != dealer['name']]
                            if other_players:
                                for _ in range(total_pot):
                                    random.choice(other_players)['score'] += 1
                            return None, -current_pot
                        else:
                            self.broadcast({'msg': f"Misfire! {first_trigger_name} got trigger-happy with no match."})
                            self.broadcast({'msg': f"Dealer {dealer['name']} takes the pot."})
                            return dealer, tie_pot
                    else:
                        high_players = self.get_high_card_winners(round_cards)
                        if len(high_players) > 1:
                            self.broadcast({'msg': f"Tie for the high card! Resolving by Thumper Power..."})
                            w, _ = self.resolve_by_thumper([p for p in tied_objs if p['name'] in high_players], dealer)
                            return w, tie_pot
                        else:
                            winner_name = high_players[0]
                            self.broadcast({'msg': f"{winner_name} holds the highest tiebreaker card!"})
                            w = next(p for p in tied_objs if p['name'] == winner_name)
                            return w, tie_pot
        finally:
            self.quickdraw_active = False
            self.quickdraw_players.clear()

    def moonshine_shootout(self, high_players, zero_players):
        self.moonshine_active = True
        self.moonshine_risk_players = high_players.copy()
        self.moonshine_zero_players = zero_players.copy()
        
        self.broadcast({'msg': "\n" + "="*40})
        self.broadcast({'msg': "!!! MOONSHINE SHOOTOUT BONUS ROUND !!!"})
        self.broadcast({'msg': "Zero-point cowboys get one last chance to steal the leader's bounty!"})
        
        risk_names = [p['name'] for p in high_players]
        zero_names = [p['name'] for p in zero_players]
        
        self.broadcast({'msg': f"Leaders (At Risk): {', '.join(risk_names)}"})
        self.broadcast({'msg': f"Zero-Pointers (Hunters): {', '.join(zero_names)}"})
        
        self.broadcast({'msg': f"MOONSHINE_BANNER|{','.join(zero_names)} vs {','.join(risk_names)}"})
        self.broadcast({'msg': "Leaders and Zero-pointers: GET READY TO SHOOT THE JUG!"})
        
        self.safe_sleep(random.uniform(3.5, 6.0))
        
        with self.lock:
            self.trigger_calls.clear()
            self.trigger_phase = True
            
        self.broadcast({'msg': "\n >>> *JUG OF MOONSHINE THROWN* CLICK CLICK! (Shoot now!) <<< \n"})
        
        participants = high_players + zero_players
        for p in participants:
            if p.get('is_ai'):
                if random.random() < 0.15: 
                    reaction = random.uniform(0.15, 0.6)
                else:
                    reaction = random.uniform(0.7, 2.4)
                threading.Timer(reaction, self.register_trigger, args=(p['name'],)).start()
                
        self.safe_sleep(JUG_AIR_TIME)
        
        with self.lock:
            self.trigger_phase = False
            self.moonshine_active = False 
            
            zero_names = [p['name'] for p in self.moonshine_zero_players]
            risk_names = [p['name'] for p in self.moonshine_risk_players]
            original_leaders = [p['name'] for p in high_players]
            
            zero_calls = [c for c in self.trigger_calls if c[1] in zero_names]
            risk_calls = [c for c in self.trigger_calls if c[1] in risk_names]
            
            # Find the fastest original leader to see if any voluntary joiners beat them
            leader_calls = [c for c in risk_calls if c[1] in original_leaders]
            leader_calls.sort(key=lambda x: x[0])
            fastest_leader_time = leader_calls[0][0] if leader_calls else float('inf')
            
            if not zero_calls:
                self.broadcast({'msg': "No zero-point player hit the jug! The leaders keep their stash."})
                self.broadcast_scores()
                return
                
            zero_calls.sort(key=lambda x: x[0])
            fastest_zero_time, fastest_zero_name = zero_calls[0]
            
            self.broadcast({'msg': f"*** {fastest_zero_name} shot the jug! ***"})
            total_stolen = 0
            stealer_pool = self.moonshine_zero_players.copy()
            
            for h in self.moonshine_risk_players:
                h_name = h['name']
                h_call = next((c for c in risk_calls if c[1] == h_name), None)
                is_voluntary = h_name not in original_leaders
                
                if not h_call or h_call[0] > fastest_zero_time:
                    stolen = h['score']
                    h['score'] = 0
                    total_stolen += stolen
                    self.broadcast({'msg': f"  -> {h_name} was too slow! They lost {stolen} cards!"})
                else:
                    self.broadcast({'msg': f"  -> {h_name} fired fast and defended their stash!"})
                    
                    # Reward voluntary risk players if they outperformed ALL leaders
                    if is_voluntary and h_call[0] < fastest_leader_time:
                        stealer_pool.append(h)
                        self.broadcast({'msg': f"  -> {h_name} outdrew the leaders and joins the hunters' split!"})
                        
            if total_stolen > 0:
                share = total_stolen // len(stealer_pool)
                remainder = total_stolen % len(stealer_pool)
                for i, p in enumerate(stealer_pool):
                    p['score'] += share + (1 if i < remainder else 0)
                
                pool_names = ", ".join([p['name'] for p in stealer_pool])
                self.broadcast({'msg': f"The stolen {total_stolen} cards are divided among: {pool_names}!"})
            else:
                self.broadcast({'msg': "The leaders survived the ambush!"})
                
            self.broadcast_scores()

    def game_loop(self, first_dealer):
        deck = []
        for _ in range(self.num_decks):
            deck.extend(create_deck())
        random.shuffle(deck)
        
        dealer_idx = self.players.index(first_dealer)
        original_dealer = first_dealer['name']
        
        # Adding check len(self.players) > 1 ensures game doesn't crash if everyone drops
        while len(deck) >= len(self.players) + 1 and len(self.players) > 1:
            dealer_idx = dealer_idx % len(self.players)
            dealer = self.players[dealer_idx]
            self.broadcast({'msg': f"\n--- ROUND {self.round_num} --- Dealer {dealer['name']} is shufflin'..."})
            self.safe_sleep(1.5)
            
            round_cards = {}
            for p in self.players:
                round_cards[p['name']] = deck.pop()
            bullseye = deck.pop()
            
            self.broadcast({'msg': f"Cards dealt face down in the dirt. The Bullseye is:"})
            self.broadcast({'msg': f"\n      >>> {bullseye['value']} of {bullseye['suit']} <<<\n"})
            
            self.safe_sleep(2)
            
            with self.lock:
                self.trigger_calls.clear()
                self.trigger_phase = True
                
            self.broadcast({'msg': f"{dealer['name']} draws and yells: 'CLICK CLICK'! (Turn 'em over!)"})
            
            cards_display = "Cards revealed on the table:\n" + "\n".join([f"  - {name}: {c['value']} of {c['suit']}" for name, c in round_cards.items()])
            self.broadcast({'msg': cards_display})
            
            match_exists = any(c['value'] == bullseye['value'] for c in round_cards.values())
            pot = len(self.players) + 1
            winner = None

            if match_exists:
                for p in self.players:
                    if p.get('is_ai'):
                        if random.random() < 0.15: 
                            reaction = random.uniform(0.15, 0.7)
                        else:
                            reaction = random.uniform(0.8, 4.8)
                        threading.Timer(reaction, self.register_trigger, args=(p['name'],)).start()

                self.safe_sleep(TRIGGER_WINDOW)
                
                initiate_quickdraw = False
                qd_ties = []
                
                with self.lock:
                    self.trigger_phase = False 
                    
                    if self.trigger_calls:
                        self.trigger_calls.sort(key=lambda x: x[0])
                        fastest_time = self.trigger_calls[0][0]
                        ties = [name for t, name in self.trigger_calls if t - fastest_time < 0.1]
                        
                        if len(ties) > 1:
                            initiate_quickdraw = True
                            qd_ties = ties
                        else:
                            winner_name = ties[0]
                            self.broadcast({'msg': f"*** {winner_name} WAS THE FASTEST GUN! ***"})
                            winner = next(p for p in self.players if p['name'] == winner_name)
                    else:
                        self.broadcast({'msg': "Nobody pulled the trigger! Let's see who's holdin' the high card..."})
                        high_players = self.get_high_card_winners(round_cards)
                        if len(high_players) > 1:
                            initiate_quickdraw = True
                            qd_ties = high_players
                        else:
                            winner_name = high_players[0]
                            self.broadcast({'msg': f"{winner_name} holds the highest card!"})
                            winner = next(p for p in self.players if p['name'] == winner_name)

                if initiate_quickdraw:
                    self.broadcast({'msg': f"Trigger TIE between {', '.join(qd_ties)}! Initiating Quickdraw Round!"})
                    winner, extra_pot = self.play_quickdraw_round(qd_ties, dealer, deck, pot)
                    pot += extra_pot

            else:
                for p in self.players:
                    if p.get('is_ai'):
                        if random.random() < 0.02: 
                            reaction = random.uniform(0.8, 3.5)
                            threading.Timer(reaction, self.register_trigger, args=(p['name'],)).start()

                self.safe_sleep(3.0) 
                
                initiate_quickdraw = False
                qd_ties = []
                
                with self.lock:
                    self.trigger_phase = False
                    if self.trigger_calls:
                        self.trigger_calls.sort(key=lambda x: x[0])
                        first_trigger_name = self.trigger_calls[0][1]
                        
                        if first_trigger_name == dealer['name']:
                            self.broadcast({'msg': f"Misfire! The Dealer ({dealer['name']}) drew on a ghost!"})
                            self.broadcast({'msg': f"The dealer loses! {pot} cards are scattered randomly to the other players."})
                            other_players = [p for p in self.players if p['name'] != dealer['name']]
                            if other_players:
                                for _ in range(pot):
                                    random.choice(other_players)['score'] += 1
                            winner = None 
                        else:
                            self.broadcast({'msg': f"Misfire! {first_trigger_name} got trigger-happy with no match."})
                            self.broadcast({'msg': f"Dealer {dealer['name']} takes the pot."})
                            winner = dealer
                    else:
                        high_players = self.get_high_card_winners(round_cards)
                        if len(high_players) > 1:
                            initiate_quickdraw = True
                            qd_ties = high_players
                        else:
                            winner_name = high_players[0]
                            self.broadcast({'msg': f"{winner_name} holds the highest card!"})
                            winner = next((p for p in self.players if p['name'] == winner_name), None)

                if initiate_quickdraw:
                    self.broadcast({'msg': f"Tie for the high card between {', '.join(qd_ties)}! Initiating Quickdraw Round!"})
                    winner, extra_pot = self.play_quickdraw_round(qd_ties, dealer, deck, pot)
                    pot += extra_pot

            if winner:
                winner['score'] += pot
                self.broadcast({'msg': f"{winner['name']} wins the round and collects {pot} cards!\n"})
            else:
                self.broadcast({'msg': "The round ends in chaos! Cards distributed.\n"})
            
            self.broadcast_scores()
            self.round_num += 1
            dealer_idx = (dealer_idx + 1) % len(self.players)
            self.safe_sleep(2.5)

        self.broadcast({'msg': "\n=== GAME OVER! The deck ran dry. ==="})
        remaining = len(deck)
        
        # Failsafe if the original dealer was kicked out mid-game
        first_dealer_obj = next((p for p in self.players if p['name'] == original_dealer), None)
        if first_dealer_obj:
            first_dealer_obj['score'] += remaining
            self.broadcast({'msg': f"First dealer {original_dealer} pockets the remaining {remaining} cards."})
            
        self.broadcast_scores()
        self.safe_sleep(2.0)
        
        if self.players:
            # Check for Moonshine Shootout
            high_score = max(p['score'] for p in self.players)
            high_players = [p for p in self.players if p['score'] == high_score]
            zero_players = [p for p in self.players if p['score'] == 0]
            
            if high_score > 0 and len(zero_players) > 0:
                self.moonshine_shootout(high_players, zero_players)
                self.safe_sleep(3.0)
            
            self.players.sort(key=lambda x: x['score'], reverse=True)
            standings = "\nFINAL BOUNTIES (Scores):\n" + "\n".join([f"  {p['name']}: {p['score']} cards" for p in self.players])
            self.broadcast({'msg': standings})
            
        self.game_active = False
        
        # FINAL BOUNTY BOARD UPDATE
        self.broadcast_scores() 
        self.broadcast({'msg': "\nHost can press 'R' to play a new game, or close the window to exit."})

class FloatingMessage:

        self.text = text
        self.color = color
        self.y = start_y

class AnimatedCard:
    def __init__(self, start_x, start_y, value=None, suit=None, label=""):
        self.x = start_x
        self.y = start_y
        self.target_x = start_x
        self.target_y = start_y
        self.draw_x = start_x
        self.draw_y = start_y
        
        self.value = value
        self.suit = suit
        self.label = label
        self.revealed = False
        
        self.bullet_holes = []  
        self.has_tombstone = False
        self.is_winner = False
        self.shake_timer = 0
        self.flash_alpha = 0

    def update(self):
        self.x = lerp(self.x, self.target_x, 0.1)
        self.y = lerp(self.y, self.target_y, 0.1)
        
        self.draw_x = self.x
        self.draw_y = self.y
        
        if self.shake_timer > 0:
            self.draw_x += random.randint(-5, 5)
            self.draw_y += random.randint(-5, 5)
            self.shake_timer -= 1
            
        if self.flash_alpha > 0:
            self.flash_alpha = max(0, self.flash_alpha - 8)

        except ValueError: pass

            if self.tumbleweed_state in ['OFF', 'BLOWING_AWAY']:
                if self.tumbleweed_state == 'OFF':
                    self.reset_tumbleweeds()
                self.tumbleweed_state = 'ROLLING'

            if self.tumbleweed_state == 'ROLLING':
                self.tumbleweed_state = 'BLOWING_AWAY'

    def validate_field(self, step_id, val):
        try:
            if step_id == 'MODE':
                if val.lower() not in ['s', 'c']: return False, "Type 'S' for Host or 'C' for Client."
                self.is_host = (val.lower() == 's')
                
                if self.is_host:
                    self.setup_sequence = ['MODE', 'NAME', 'AGE', 'HEIGHT', 'WEIGHT', 'DECKS', 'NETWORK', 'PORT']
                else:
                    self.setup_sequence = ['MODE', 'NAME', 'AGE', 'HEIGHT', 'WEIGHT', 'NETWORK', 'PORT']
                    
            elif step_id == 'NAME':
                if not val.strip(): return False, "Name cannot be empty."
                self.name = val.strip()
            elif step_id == 'AGE':
                v = float(val)
                if v <= 0: return False, "Age must be > 0."
                self.age = v
            elif step_id == 'HEIGHT':
                v = float(val)
                if v <= 0: return False, "Height must be > 0."
                self.height = v
            elif step_id == 'WEIGHT':
                v = float(val)
                if v <= 0: return False, "Weight must be > 0."
                self.weight_lbs = v
                self.thumper = calculate_thumper(self.age, self.height, self.weight_lbs * 0.453592)
            elif step_id == 'DECKS':
                v = int(val)
                if not (1 <= v <= 5): return False, "Decks must be 1-5."
                self.num_decks = v
            elif step_id == 'NETWORK':
                if self.is_host:
                    v = int(val)
                    if not (0 <= v <= 7): return False, "AI count 0-7."
                    self.ai_count = v
                else:
                    if not val.strip(): return False, "IP cannot be empty."
                    self.host_ip = val.strip()
            elif step_id == 'PORT':
                v = int(val)
                if not (1024 < v <= 65535): return False, "Port 1025-65535."
                self.port = v
            return True, ""
        except ValueError:
            return False, "Invalid format. Numbers required here."

    def handle_setup_input(self, event):
        current_field = self.setup_sequence[self.setup_idx]
        
        if event.key == pygame.K_BACKSPACE:
            self.setup_data[current_field] = self.setup_data[current_field][:-1]
            self.setup_error = ""
        elif event.key == pygame.K_UP:
            if self.setup_idx > 0:
                self.setup_idx -= 1
                self.setup_error = ""
                self.evaluate_tumbleweeds()
        elif event.key == pygame.K_DOWN:
            valid, err = self.validate_field(current_field, self.setup_data[current_field])
            if valid:
                if self.setup_idx < len(self.setup_sequence) - 1:
                    self.setup_idx += 1
                    self.setup_error = ""
                    self.evaluate_tumbleweeds()
            else:
                self.setup_error = err
        elif event.key in (pygame.K_RETURN, pygame.K_KP_ENTER):
            valid, err = self.validate_field(current_field, self.setup_data[current_field])
            if valid:
                if self.setup_idx < len(self.setup_sequence) - 1:
                    self.setup_idx += 1
                    self.setup_error = ""
                    self.evaluate_tumbleweeds()
                else:
                    all_valid = True
                    for i, f in enumerate(self.setup_sequence):
                        v, e = self.validate_field(f, self.setup_data[f])
                        if not v:
                            self.setup_idx = i
                            self.setup_error = e
                            all_valid = False
                            break
                    if all_valid:
                        self.finalize_setup()
            else:
                self.setup_error = err
        else:
            if event.unicode.isprintable() and len(self.setup_data[current_field]) < 20:
                self.setup_data[current_field] += event.unicode
                self.setup_error = ""

    def finalize_setup(self):
        pygame.display.set_caption("TRIGGERED - High Res Edition" + (" [HOST]" if self.is_host else ""))
        
        if self.is_host:
            try:
                ai_qty = int(self.setup_data.get('NETWORK', '0'))
            except:
                ai_qty = 0
            self.ai_count = ai_qty
            
            self.server = TriggeredServer(self.port, self.ai_count, self.num_decks)
            self.server.start()
            time.sleep(0.5)
            self.host_ip = '127.0.0.1'
            
        self.connect_to_server()

    def send_heartbeats(self):
        """ Keeps the client connection alive by pinging the server """
        while self.running and self.client_socket:
            try:
                msg = json.dumps({'action': 'ping'}) + "\n"
                self.client_socket.sendall(msg.encode())
            except:
                pass
            time.sleep(5)

    def connect_to_server(self):
        self.client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        try:
            self.client_socket.connect((self.host_ip, self.port))
            
            self.state = 'TITLE'
            
            # Start threads first to prevent missing the initial score broadcast
            threading.Thread(target=self.receive_messages, daemon=True).start()
            
            # Send info WITH A NEWLINE so the server buffers it correctly!
            info = json.dumps({'name': self.name, 'thumper': self.thumper}) + "\n"
            self.client_socket.sendall(info.encode())
            
            # Start heartbeats AFTER sending info to guarantee ordering
            threading.Thread(target=self.send_heartbeats, daemon=True).start()
            
            self.add_log_message("Welcome to the Saloon!", SVGA['TEXT_LIGHT'])
            self.log_game_text("Welcome to the Saloon!")
            
            if self.is_host:
                self.add_log_message("You are the Host. Press 'S' to Start.", SVGA['GOLD'])
                self.log_game_text("You are the Host. Press 'S' to Start.")
            else:
                self.add_log_message("Waiting for the Host to start...", SVGA['TEXT_LIGHT'])
                self.log_game_text("Waiting for the Host to start...")
                
            self.add_log_message("Press 'T' to pull the TRIGGER!", SVGA['TEXT_LIGHT'])
            self.log_game_text("Press 'T' to pull the TRIGGER!")
            self.add_log_message("Press 'Q' to toggle audio.", SVGA['TEXT_LIGHT'])
            self.log_game_text("Press 'Q' to toggle audio.")
            self.add_log_message("Press 'P' to Pause.", SVGA['TEXT_LIGHT'])
            self.log_game_text("Press 'P' to Pause.")
            
        except ConnectionRefusedError:
            self.setup_idx = self.setup_sequence.index('PORT')
            self.setup_error = "Connection Failed. Check IP/Port and try again."

    def receive_messages(self):
        buffer = ""
        while self.running and self.state in ['TITLE', 'DOOR_ANIM', 'GAME']:
            try:
                data = self.client_socket.recv(2048).decode()
                if not data: break
                buffer += data
                
                while "\n" in buffer:
                    line, buffer = buffer.split("\n", 1)
                    line = line.strip()
                    if not line: continue
                    
                    try:
                        msg = json.loads(line)
                        
                        if isinstance(msg, dict):
                            if msg.get('action') == 'update_scores':
                                self.current_scores = msg.get('scores', {})
                                continue
                                
                            if msg.get('action') == 'remove_player':
                                name_to_remove = msg.get('name')
                                if name_to_remove in self.player_cards:
                                    del self.player_cards[name_to_remove]
                                if name_to_remove in self.all_player_names:
                                    self.all_player_names.remove(name_to_remove)
                                continue
                                
                            if 'msg' in msg:
                                raw_msg = msg['msg']
                                
                                if "MOONSHINE_BANNER|" in raw_msg:
                                    parts = raw_msg.split("|")[1].split(" vs ")
                                    self.moonshine_hunters = parts[0].split(",")
                                    self.moonshine_leaders = parts[1].split(",")
                                    self.moonshine_waiting = True
                                    self.show_moonshine_jug = False
                                    self.jug_shot = False
                                    self.player_cards.clear()
                                    self.bullseye_anim_card = None
                                    continue 
                                
                                if "QUICKDRAW_BANNER|" in raw_msg:
                                    self.quickdraw_tied_players = raw_msg.split("|")[1].split(",")
                                    self.quickdraw_waiting = True
                                    self.quickdraw_timer = 120  
                                    continue

                                self.log_game_text(raw_msg)
                                
                                if "GAME PAUSED" in raw_msg:
                                    self.is_paused = True
                                elif "GAME RESUMED" in raw_msg:
                                    self.is_paused = False
                                    
                                elif "DRAW YOUR WEAPONS!" in raw_msg:
                                    self.game_over = False
                                    self.overall_winner = ""
                                    self.is_quickdraw = False
                                    
                                elif "=== GAME OVER! The deck ran dry. ===" in raw_msg:
                                    self.player_cards.clear()
                                    self.bullseye_anim_card = None
                                    self.cards_dealt = False
                                    self.is_quickdraw = False
                                    
                                elif "is shufflin'" in raw_msg:
                                    if "SPECIAL QUICKDRAW" in raw_msg:
                                        self.is_quickdraw = True
                                    else:
                                        self.is_quickdraw = False
                                        
                                    self.bullseye_anim_card = None
                                    self.show_moonshine_jug = False
                                    for p_name, card in self.player_cards.items():
                                        card.revealed = False
                                        card.bullet_holes.clear()
                                        card.is_winner = False 
                                        card.has_tombstone = False
                                        card.x = self.res[0] + 200 
                                        card.target_x = self.res[0] + 200
                                        card.y = self.res[1] + 200

                                elif "--- THUMPER POWERS ---" in raw_msg:
                                    self.player_cards.clear()
                                    self.all_player_names = []
                                    idx = 0
                                    for m_line in raw_msg.split('\n'):
                                        if ':' in m_line and not '---' in m_line and not 'THUMPER' in m_line:

                                                idx += 1

                                elif any(phrase in raw_msg for phrase in ["*BANG*", "shot at the jug!", "*Click*", "Misfire!"]):
                                    shooter = None

                                            break

                                        self.player_cards[winner_match].shake_timer = 40

                                    start_x = self.left_panel_w + 50
                                    for i, (p_name, card) in enumerate(self.player_cards.items()):
                                        col = i % 4

            pygame.draw.rect(flash_surf, fc, (lx - 30, ly - 60, 60, 90))
            pygame.draw.arc(flash_surf, fc, (lx + 25, ly - 40, 50, 60), -math.pi/2, math.pi/2, 12)
            
            if not self.jug_shot:
                pygame.draw.rect(flash_surf, fc, (lx - 20, ly - 80, 40, 20))
            
            self.screen.blit(flash_surf, (cx - 100, cy - 100))
            self.jug_flash_alpha = max(0, self.jug_flash_alpha - 5)

        # Draw Moonshine Splash Particles
        for p in reversed(self.moonshine_particles):
            p['x'] += p['dx']
            p['y'] += p['dy']
            p['dy'] += 0.4 # Liquid gravity
            p['life'] -= 5
            
            if p['life'] <= 0:
                self.moonshine_particles.remove(p)
            else:
                alpha = max(0, min(255, int(p['life'])))
                p_color = (200, 220, 255, alpha)
                psurf = pygame.Surface((int(p['size']*2), int(p['size']*2)), pygame.SRCALPHA)
                pygame.draw.circle(psurf, p_color, (int(p['size']), int(p['size'])), int(p['size']))
                self.screen.blit(psurf, (int(p['x'] - p['size']), int(p['y'] - p['size'])))

        # Draw Cork (Top Layer)
        if self.jug_shot:
            cork_draw_x = self.cork_world_x
            cork_draw_y = self.cork_world_y
        else:
            cork_draw_x = cx - 20
            cork_draw_y = cy - 80
            
        pygame.draw.rect(self.screen, (200, 170, 120), (cork_draw_x, cork_draw_y, 40, 20))
        pygame.draw.rect(self.screen, (120, 90, 50), (cork_draw_x, cork_draw_y, 40, 20), 3)

    def draw_card(self, x, y, width, height, value, suit, label, revealed, is_winner=False, bullet_holes=None, has_tombstone=False, flash_alpha=0):
        if bullet_holes is None:
            bullet_holes = []
            
        lbl_color = SVGA['TEXT_GREEN'] if is_winner else SVGA['GOLD']

        pygame.draw.rect(shadow_rect, SVGA['CARD_SHADOW'], shadow_rect.get_rect(), border_radius=8)

        border_thickness = 4 if is_winner else 2
        
        if not revealed:
            pygame.draw.rect(self.screen, (20, 50, 100), (x, y, width, height), border_radius=8)
            pygame.draw.rect(self.screen, border_color, (x, y, width, height), border_thickness, border_radius=8)
            for i in range(10, width, 15):
                pygame.draw.line(self.screen, (30, 80, 150), (x+i, y), (x+i, y+height), 2)
        else:
            pygame.draw.rect(self.screen, SVGA['WHITE'], (x, y, width, height), border_radius=8)
            pygame.draw.rect(self.screen, border_color, (x, y, width, height), border_thickness, border_radius=8) 
            
            color = SVGA['SUIT_RED'] if suit in ['Hearts', 'Diamonds'] else SVGA['SUIT_BLACK']
            
            val_surf = self.card_font.render(value, True, color)
            self.screen.blit(val_surf, (x + 10, y + 10))
            
            cx, cy = x + width // 2, y + height // 2 + 10
            size = 20 if width > 100 else 12
            
            if suit == 'Diamonds':
                pygame.draw.polygon(self.screen, color, [(cx, cy-size), (cx+size*0.8, cy), (cx, cy+size), (cx-size*0.8, cy)])
            elif suit == 'Hearts':
                pygame.draw.circle(self.screen, color, (cx-size//2, cy-size//2), size//2 + 2)
                pygame.draw.circle(self.screen, color, (cx+size//2, cy-size//2), size//2 + 2)
                pygame.draw.polygon(self.screen, color, [(cx-size, cy-size//3), (cx+size, cy-size//3), (cx, cy+size)])
            elif suit == 'Spades':
                pygame.draw.circle(self.screen, color, (cx-size//2, cy+size//3), size//2 + 2)
                pygame.draw.circle(self.screen, color, (cx+size//2, cy+size//3), size//2 + 2)
                pygame.draw.polygon(self.screen, color, [(cx-size, cy+size//3), (cx+size, cy+size//3), (cx, cy-size)])
                pygame.draw.rect(self.screen, color, (cx-size//4, cy+size//3, size//2, size))
            elif suit == 'Clubs':
                pygame.draw.circle(self.screen, color, (cx, cy-size//1.5), size//2 + 2)
                pygame.draw.circle(self.screen, color, (cx-size//1.5, cy+size//3), size//2 + 2)
                pygame.draw.circle(self.screen, color, (cx+size//1.5, cy+size//3), size//2 + 2)
                pygame.draw.rect(self.screen, color, (cx-size//4, cy+size//3, size//2, size))

        if flash_alpha > 0:
            flash_surf = pygame.Surface((width + 40, height + 40), pygame.SRCALPHA)
            pygame.draw.rect(flash_surf, (255, 255, 255, flash_alpha), flash_surf.get_rect(), border_radius=16)
            self.screen.blit(flash_surf, (x - 20, y - 20))

        for bx_offset, by_offset in bullet_holes:
            bx, by = x + width // 2 + bx_offset, y + height // 2 + by_offset
            pygame.draw.circle(self.screen, (70, 70, 70), (bx, by), 12)
            pygame.draw.circle(self.screen, (15, 15, 15), (bx, by), 7)
            pygame.draw.line(self.screen, (40, 40, 40), (bx, by), (bx+18, by-12), 2)
            pygame.draw.line(self.screen, (40, 40, 40), (bx, by), (bx-15, by-15), 2)
            pygame.draw.line(self.screen, (40, 40, 40), (bx, by), (bx-8, by+18), 2)

        if has_tombstone:
            tx = x + width // 2
            ty = y + height // 2 + 10
            
            pygame.draw.rect(self.screen, (100, 100, 100), (tx - 30, ty - 10, 60, 50))
            pygame.draw.circle(self.screen, (100, 100, 100), (tx, ty - 10), 30)
            
            pygame.draw.line(self.screen, (50, 50, 50), (tx - 30, ty - 10), (tx - 30, ty + 38), 2) 
            pygame.draw.line(self.screen, (50, 50, 50), (tx + 30, ty - 10), (tx + 30, ty + 38), 2) 
            pygame.draw.line(self.screen, (50, 50, 50), (tx - 31, ty + 39), (tx + 31, ty + 39), 2) 
            pygame.draw.arc(self.screen, (50, 50, 50), (tx - 30, ty - 40, 60, 60), 0, math.pi, 2)  
            
            rip_surf = self.small_font.render("R.I.P.", True, SVGA['BLACK'])
            self.screen.blit(rip_surf, (tx - rip_surf.get_width()//2, ty - 5))

    def draw_game_screen(self):

        for msg in self.log_messages:

        # --- Draw Score Board ---
        if self.current_scores and self.state == 'GAME':
            board_w = 200

            sorted_scores = sorted(self.current_scores.items(), key=lambda x: x[1], reverse=True)
            y_off = by + 40
            for name, score in sorted_scores:
                color = SVGA['TEXT_GREEN'] if name == self.name else SVGA['WHITE']
                name_str = (name[:12] + '..') if len(name) > 14 else name
                entry = self.small_font.render(f"{name_str}: {score}", True, color)
                self.screen.blit(entry, (bx + 15, y_off))
                y_off += 25

        if not self.game_over:
            if self.moonshine_waiting:
                self.draw_moonshine_waiting_sign()
            elif self.show_moonshine_jug:
                self.draw_moonshine_jug()
            else:
                if self.bullseye_anim_card:
                    c = self.bullseye_anim_card
                    c.update()
                    self.draw_card(c.draw_x, c.draw_y, 150, 210, c.value, c.suit, c.label, c.revealed, bullet_holes=c.bullet_holes)
                    
                    if self.is_quickdraw and c.revealed:
                        self.draw_pistol(c.draw_x - 40, c.draw_y + 105, flip_x=False)
                        self.draw_pistol(c.draw_x + 190, c.draw_y + 105, flip_x=True)
                    
                for p_name, c in self.player_cards.items():
                    c.update()
                    label = "YOU" if p_name == self.name else p_name
                    self.draw_card(c.draw_x, c.draw_y, 100, 140, c.value, c.suit, label, c.revealed, c.is_winner, c.bullet_holes, c.has_tombstone, c.flash_alpha)

            if self.quickdraw_waiting:
                self.draw_quickdraw_waiting_sign()
                self.quickdraw_timer -= 1
                if self.quickdraw_timer <= 0:
                    self.quickdraw_waiting = False

        if self.game_over and self.overall_winner:
            font_big = pygame.font.SysFont('impact', 64)
            win_text_str = f"WINNER: {self.overall_winner}"

            board_h = 180
            cx = self.left_panel_w + (self.res[0] - self.left_panel_w)//2
            cy = self.res[1]//2

    def process_messages(self):
        while not self.msg_queue.empty():
            text, color = self.msg_queue.get()
            if text == "":
                self.add_log_message(" ", color)
            else:
                self.add_log_message(text, color)

    def run(self):
        clock = pygame.time.Clock()
        
        while self.running:
            self.process_messages()
            
            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    self.running = False
                    if self.is_host and self.client_socket:
                        msg = json.dumps({'action': 'exit'}) + "\n"
                        try:
                            self.client_socket.sendall(msg.encode())
                        except:
                            pass
                elif event.type == pygame.KEYDOWN:
                    
                    # 1. THE GLOBAL QUIET BUTTON
                    if event.key == pygame.K_q:
                        self.sound_enabled = not self.sound_enabled
                        if self.sound_enabled:
                            try: pygame.mixer.music.set_volume(0.3)
                            except: pass
                            self.add_log_message("Sound: ON", SVGA['TEXT_GREEN'])
                            self.log_game_text("Sound: ON")
                        else:
                            try: pygame.mixer.music.set_volume(0.0)
                            except: pass
                            self.add_log_message("Sound: MUTED", SVGA['TEXT_RED'])
                            self.log_game_text("Sound: MUTED")
                        continue 
                        
                    # SETUP STATE LOGIC
                    if self.state == 'SETUP':
                        self.handle_setup_input(event)
                        continue
                        
                    # 2. GLOBAL NETWORK COMMANDS (Available in TITLE, DOOR_ANIM, and GAME states)
                    if event.key == pygame.K_p:
                        msg = json.dumps({'action': 'toggle_pause'}) + "\n"
                        try: self.client_socket.sendall(msg.encode())
                        except: pass
                        
                    elif event.key == pygame.K_s:
                        if self.is_host:
                            msg = json.dumps({'action': 'start'}) + "\n"
                            try: self.client_socket.sendall(msg.encode())
                            except: pass
                        else:
                            self.add_log_message("Only the Host can start the game.", SVGA['TEXT_RED'])
                            self.log_game_text("Only the Host can start the game.")
                            
                    elif event.key == pygame.K_r:
                        if self.is_host:
                            msg = json.dumps({'action': 'restart'}) + "\n"
                            try: self.client_socket.sendall(msg.encode())
                            except: pass
                        else:
                            self.add_log_message("Only the Host can restart the game.", SVGA['TEXT_RED'])
                            self.log_game_text("Only the Host can restart the game.")

                    # 3. STATE-SPECIFIC COMMANDS
                    if self.state == 'TITLE':
                        if event.key == pygame.K_SPACE:
                            self.state = 'DOOR_ANIM' 
                            self.door_progress = 0.0
                            
                        elif event.key == pygame.K_d:
                            if not self.debug_logging:
                                self.debug_logging = True
                                self.debug_filename = f"debug_{int(time.time())}.log"
                                try:
                                    with open(self.debug_filename, 'w') as f:
                                        f.write("--- TRIGGERED RULES & TEXT DUMP ---\n\nRULES:\n")
                                        rules = [
                                            "1. Goal: Collect the most cards, partner.",
                                            "2. A 'Bullseye' card is dealt face-up.",
                                            "3. When dealer yells 'CLICK CLICK', cards flip.",
                                            "4. QUICK DRAW: If ANY match to Bullseye, press the 'T' key immediately!",
                                            "5. No match? Highest card takes the pot.",
                                            "6. TIE-BREAKER: Ties trigger a sudden-death Quickdraw round!",
                                            "7. MISFIRE: Press 'T' with no match? You lose the round!",
                                            "8. MOONSHINE ROUND: Zeros can steal from the leader in the end!"
                                        ]
                                        f.write("\n".join(rules) + "\n\nMESSAGE HISTORY:\n")
                                        f.write("\n".join(self.full_message_history) + "\n")
                                    print(f"Successfully started logging game text to {self.debug_filename}")
                                except Exception as e:
                                    print(f"Failed to start debug log file: {e}")
                            
                    elif self.state == 'GAME':
                        if event.key == pygame.K_t: 
                            if self.show_moonshine_jug:
                                if self.jug_y < self.res[1] + 150:
                                    if not self.jug_shot:
                                        cx = self.left_panel_w + (self.res[0] - self.left_panel_w)//2
                                        self.cork_world_x = cx - 20
                                        self.cork_world_y = self.jug_y - 80
                                        self.cork_dx = random.choice([-6, 6])
                                        self.cork_dy = self.jug_dy - 15
                                        
                                    self.jug_bullet_holes.append((random.randint(-40, 40), random.randint(-40, 60)))
                                    self.jug_flash_alpha = 255
                                    self.jug_shot = True
                                    
                                    cx = self.left_panel_w + (self.res[0] - self.left_panel_w)//2
                                    for _ in range(random.randint(20, 35)):
                                        self.moonshine_particles.append({
                                            'x': cx,
                                            'y': self.jug_y - 60,
                                            'dx': random.uniform(-8, 8),
                                            'dy': random.uniform(-15, -5),
                                            'life': random.uniform(200, 255),
                                            'size': random.uniform(3, 7)
                                        })
                                    
                                    if self.gunshot_sound and self.sound_enabled:
                                        self.gunshot_sound.play()
                                        
                                    msg = json.dumps({'action': 'trigger'}) + "\n"
                                    try: self.client_socket.sendall(msg.encode())
                                    except: pass
                            else:
                                self.add_log_message(">>> You reached for your iron! <<<", SVGA['TEXT_BLUE'])
                                self.log_game_text(">>> You reached for your iron! <<<")
                                
                                if self.gunshot_sound and self.sound_enabled:
                                    self.gunshot_sound.play()
                                    
                                if self.cards_dealt and self.name in self.player_cards:
                                    c = self.player_cards[self.name]
                                    c.bullet_holes.append((random.randint(-35, 35), random.randint(-45, 45)))
                                    c.flash_alpha = 255
                                    
                                msg = json.dumps({'action': 'trigger'}) + "\n"
                                try: self.client_socket.sendall(msg.encode())
                                except: pass

            if self.state == 'SETUP':
                self.draw_setup_screen()
            elif self.state == 'TITLE':
                self.draw_title_screen()
            elif self.state == 'DOOR_ANIM':
                self.draw_door_anim_screen()
            elif self.state == 'GAME':
                self.draw_game_screen()
                
            clock.tick(60) 
            
        pygame.quit()
        os._exit(0)

if __name__ == "__main__":
    app = TriggeredGameApp()
    app.run()
