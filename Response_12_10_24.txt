```python
import socket
import threading
import json
import time
import random
import sys
import os
import queue
import math
import pygame

# --- Constants & Colors ---
SVGA = {
    'BG_GREEN': (24, 66, 38),      
    'BLACK': (20, 20, 20),
    'WHITE': (245, 245, 245),
    'GOLD': (255, 215, 0),
    'TEXT_RED': (255, 100, 100),
    'POSTER': (222, 199, 155),
    'INK': (45, 35, 25)
}

# --- Tumbleweed Object ---
class Tumbleweed:
    def __init__(self, screen_w, screen_y):
        self.x = -100
        self.y = screen_y
        self.angle = 0
        self.speed = random.uniform(3, 6)

    def update(self):
        self.x += self.speed
        self.angle += self.speed * 2
        self.y += math.sin(self.x * 0.04) * 4

    def draw(self, screen):
        color = (120, 90, 50)
        cx, cy = int(self.x), int(self.y)
        for i in range(5):
            a = math.radians(self.angle + (i * 72))
            pygame.draw.line(screen, color, (cx, cy), (cx + math.cos(a)*30, cy + math.sin(a)*30), 2)
        pygame.draw.circle(screen, color, (cx, cy), 28, 2)

# --- Main App ---
class TriggeredGameApp:
    def __init__(self):
        pygame.init()
        self.res = (1280, 720)
        self.screen = pygame.display.set_mode(self.res)
        pygame.display.set_caption("TRIGGERED - High Res Edition")
        
        self.font = pygame.font.SysFont('courier', 28, bold=True)
        self.title_font = pygame.font.SysFont('impact', 80)
        
        self.state = 'SETUP'
        self.setup_step = 0
        self.input_text = ""
        self.error = ""
        self.params = {'MODE': '', 'NAME': '', 'AGE': 0, 'HEIGHT': 0, 'WEIGHT': 0, 'DECKS': 1, 'PORT': 5555}
        self.steps = ['MODE', 'NAME', 'AGE', 'HEIGHT', 'WEIGHT', 'DECKS', 'PORT']
        
        self.tumbleweeds = []
        self.running = True

    def check_extremes(self, key, val):
        try:
            val = float(val)
            if (key == 'AGE' and val > 110) or (key == 'HEIGHT' and val > 8) or (key == 'WEIGHT' and val > 450):
                self.tumbleweeds.append(Tumbleweed(self.res[0], self.res[1] - 80))
        except: pass

    def handle_setup(self, event):
        if event.key == pygame.K_BACKSPACE:
            self.input_text = self.input_text[:-1]
        elif event.key in (pygame.K_RETURN, pygame.K_KP_ENTER):
            key = self.steps[self.setup_step]
            val = self.input_text.strip()
            
            if key == 'MODE':
                if val.upper() in ('S', 'C'):
                    self.params[key] = val.upper()
                    self.setup_step += 1
                else: self.error = "Enter 'S' (Host) or 'C' (Client)"
            elif key == 'NAME':
                if val: 
                    self.params[key] = val
                    self.setup_step += 1
                else: self.error = "Name required!"
            elif key in ('AGE', 'HEIGHT', 'WEIGHT', 'DECKS', 'PORT'):
                try:
                    num = float(val) if key == 'HEIGHT' else int(val)
                    if num <= 0: raise ValueError
                    
                    if key == 'PORT':
                        if 1024 <= num <= 65535:
                            self.params[key] = num
                            self.state = 'TITLE' # Proceed to Game
                        else: self.error = "Invalid Port (1024-65535)"
                    else:
                        self.check_extremes(key, val)
                        self.params[key] = num
                        self.setup_step += 1
                except: self.error = "Must be a positive number!"
            
            self.input_text = ""
            if not self.error: self.error = ""

    def draw_wanted_poster(self):
        self.screen.fill(SVGA['BG_GREEN'])
        for tw in self.tumbleweeds[:]:
            tw.update()
            tw.draw(self.screen)
            if tw.x > self.res[0] + 100: self.tumbleweeds.remove(tw)

        rect = pygame.Rect(self.res[0]//2 - 350, 40, 700, 640)
        pygame.draw.rect(self.screen, SVGA['POSTER'], rect)
        pygame.draw.rect(self.screen, SVGA['INK'], rect, 5)
        
        title = self.title_font.render("WANTED", True, SVGA['INK'])
        self.screen.blit(title, (self.res[0]//2 - title.get_width()//2, 60))
        
        prompts = {
            'MODE': "ROLE: (S)ERVER HOST OR (C)LIENT?",
            'NAME': "YOUR NAME, PARTNER?",
            'AGE': "HOW MANY YEARS OLD?",
            'HEIGHT': "HOW TALL (FEET)?",
            'WEIGHT': "WEIGHT (LBS)?",
            'DECKS': "HOW MANY DECKS OF CARDS?",
            'PORT': "WHICH PORT (e.g. 5555)?"
        }
        
        curr_key = self.steps[self.setup_step]
        txt = self.font.render(prompts[curr_key], True, SVGA['INK'])
        self.screen.blit(txt, (self.res[0]//2 - txt.get_width()//2, 250))
        
        inp = self.font.render("> " + self.input_text + "_", True, (180, 0, 0))
        self.screen.blit(inp, (self.res[0]//2 - inp.get_width()//2, 320))
        
        if self.error:
            err_txt = self.font.render(self.error, True, SVGA['TEXT_RED'])
            self.screen.blit(err_txt, (self.res[0]//2 - err_txt.get_width()//2, 580))

    def run(self):
        clock = pygame.time.Clock()
        while self.running:
            for event in pygame.event.get():
                if event.type == pygame.QUIT: self.running = False
                if event.type == pygame.KEYDOWN and self.state == 'SETUP':
                    self.handle_setup(event)
                elif event.type == pygame.KEYDOWN and self.state == 'TITLE':
                    if event.key == pygame.K_SPACE: self.running = False # Exit to shell or next stage

            if self.state == 'SETUP':
                self.draw_wanted_poster()
            else:
                self.screen.fill(SVGA['BLACK'])
                msg = self.font.render("READY. PRESS SPACE TO START SERVER.", True, SVGA['GOLD'])
                self.screen.blit(msg, (self.res[0]//2 - msg.get_width()//2, self.res[1]//2))
            
            pygame.display.flip()
            clock.tick(60)
        pygame.quit()

if __name__ == "__main__":
    app = TriggeredGameApp()
    app.run()
```
