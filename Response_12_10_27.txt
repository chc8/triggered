                            winner_name = high_players[0]
                            self.broadcast({'msg': f"{winner_name} holds the highest card!"})
                            winner = next(p for p in self.players if p['name'] == winner_name)

            if winner:
                winner['score'] += pot
                self.broadcast({'msg': f"{winner['name']} wins the round and collects {pot} cards!\n"})
            else:
                self.broadcast({'msg': "The round ends in chaos! Cards distributed.\n"})
            
            self.round_num += 1
            dealer_idx = (dealer_idx + 1) % len(self.players)
            self.safe_sleep(2.5)

        self.broadcast({'msg': "\n=== GAME OVER! The deck ran dry. ==="})
        remaining = len(deck)
        first_dealer_obj = next(p for p in self.players if p['name'] == original_dealer)
        first_dealer_obj['score'] += remaining
        self.broadcast({'msg': f"First dealer {original_dealer} pockets the remaining {remaining} cards."})
        self.safe_sleep(2.0)
        
        high_score = max(p['score'] for p in self.players)
        high_players = [p for p in self.players if p['score'] == high_score]
        zero_players = [p for p in self.players if p['score'] == 0]
        
        if high_score > 0 and len(zero_players) > 0:
            self.moonshine_shootout(high_players, zero_players)
            self.safe_sleep(3.0)
        
        self.players.sort(key=lambda x: x['score'], reverse=True)
        standings = "\nFINAL BOUNTIES (Scores):\n" + "\n".join([f"  {p['name']}: {p['score']} cards" for p in self.players])
        self.broadcast({'msg': standings})
        
        self.game_active = False
        self.broadcast({'msg': "\nHost can press 'R' to play a new game, or close the window to exit."})

# --- Visual Objects ---
class FloatingMessage:
    def __init__(self, text, color, start_y):
        self.text = text
        self.color = color
        self.y = start_y
        self.target_y = start_y

    def update(self):
        self.y = lerp(self.y, self.target_y, 0.15)

class AnimatedCard:
    def __init__(self, start_x, start_y, value=None, suit=None, label=""):
        self.x = start_x
        self.y = start_y
        self.target_x = start_x
        self.target_y = start_y
        self.draw_x = start_x
        self.draw_y = start_y
        
        self.value = value
        self.suit = suit
        self.label = label
        self.revealed = False
        
        self.bullet_holes = []  
        self.has_tombstone = False
        self.is_winner = False
        self.shake_timer = 0
        self.flash_alpha = 0

    def update(self):
        self.x = lerp(self.x, self.target_x, 0.1)
        self.y = lerp(self.y, self.target_y, 0.1)
        
        self.draw_x = self.x
        self.draw_y = self.y
        
        if self.shake_timer > 0:
            self.draw_x += random.randint(-5, 5)
            self.draw_y += random.randint(-5, 5)
            self.shake_timer -= 1
            
        if self.flash_alpha > 0:
            self.flash_alpha = max(0, self.flash_alpha - 8)

    def check_extremes(self, key, val):
        """Checks if a value is biologically extreme and triggers tumbleweeds."""
        try:
            v = float(val)
            # Thresholds for '2 standard deviations' (Approximate for fun)
            extreme = False
            if key == 'AGE' and (v > 105 or v < 5): extreme = True
            if key == 'HEIGHT' and (v > 7.5 or v < 3.5): extreme = True
            if key == 'WEIGHT' and (v > 400 or v < 60): extreme = True
            
            if extreme:
                for _ in range(random.randint(2, 4)):
                    self.tumbleweeds.append(Tumbleweed(self.res[0], self.res[1] - 80))
        except: pass

    def handle_setup_input(self, event):
        if event.key == pygame.K_BACKSPACE:
            self.input_text = self.input_text[:-1]
            self.setup_error = ""
        elif event.key in (pygame.K_RETURN, pygame.K_KP_ENTER):
            val = self.input_text.strip()
            self.input_text = ""
            self.setup_error = ""
            
            try:
                if self.setup_step == 'MODE':
                    if val.lower() == 's':
                        self.is_host = True
                        self.setup_step = 'NAME'
                    elif val.lower() == 'c':
                        self.is_host = False
                        self.setup_step = 'NAME'
                    else:
                        self.setup_error = "Type 'S' for Host or 'C' for Client."
                
                elif self.setup_step == 'NAME':
                    if val:
                        self.name = val
                        self.setup_step = 'AGE'
                    else:
                        self.setup_error = "Name cannot be empty."
                        
                elif self.setup_step == 'AGE':
                    v = float(val)
                    if v <= 0: self.setup_error = "Age must be greater than zero."
                    else:
                        self.check_extremes('AGE', v)
                        self.age = v
                        self.setup_step = 'HEIGHT'
                    
                elif self.setup_step == 'HEIGHT':
                    v = float(val)
                    if v <= 0: self.setup_error = "Height must be greater than zero."
                    else:
                        self.check_extremes('HEIGHT', v)
                        self.height = v
                        self.setup_step = 'WEIGHT'
                    
                elif self.setup_step == 'WEIGHT':
                    v = float(val)
                    if v <= 0: self.setup_error = "Weight must be greater than zero."
                    else:
                        self.check_extremes('WEIGHT', v)
                        self.weight_lbs = v
                        weight_kg = self.weight_lbs * 0.453592
                        self.thumper = calculate_thumper(self.age, self.height, weight_kg)
                        self.setup_step = 'NETWORK'
                    
                elif self.setup_step == 'NETWORK':
                    if self.is_host:
                        self.ai_count = int(val)
                        if 0 <= self.ai_count <= 7:
                            self.setup_step = 'PORT'
                            self.input_text = "5555" 
                        else:
                            self.setup_error = "AI count must be 0-7."
                    else:
                        if val:
                            self.host_ip = val
                            self.setup_step = 'PORT'
                            self.input_text = "5555" 
                        else:
                            self.setup_error = "IP cannot be empty."
                        
                elif self.setup_step == 'PORT':
                    p_val = int(val)
                    if p_val < 1024 or p_val > 65535:
                        self.setup_error = "Port reserved or invalid! Use 1024-65535."
                    else:
                        self.port = p_val
                        self.finalize_setup()
                    
            except ValueError:
                self.setup_error = "Invalid format. Positive numbers required."
        else:
            if event.unicode.isprintable() and len(self.input_text) < 20:
                self.input_text += event.unicode
                self.setup_error = ""

    def finalize_setup(self):

        self.connect_to_server()

    def connect_to_server(self):
        self.client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        try:
            self.client_socket.connect((self.host_ip, self.port))
            info = json.dumps({'name': self.name, 'thumper': self.thumper})
            self.client_socket.sendall(info.encode())
            
            self.state = 'TITLE'
            
            self.add_log_message("Welcome to the Saloon!", SVGA['TEXT_LIGHT'])
            self.log_game_text("Welcome to the Saloon!")
            
            if self.is_host:
                self.add_log_message(f"Server IP: {get_local_ip()} | Port: {self.port}", SVGA['TEXT_BLUE'])
                self.log_game_text(f"Server IP: {get_local_ip()} | Port: {self.port}")
                self.add_log_message("You are the Host. Press 'S' to Start.", SVGA['GOLD'])
                self.log_game_text("You are the Host. Press 'S' to Start.")
            else:
                self.add_log_message("Waiting for the Host to start...", SVGA['TEXT_LIGHT'])
                self.log_game_text("Waiting for the Host to start...")
                
            threading.Thread(target=self.receive_messages, daemon=True).start()
        except Exception:
            self.setup_step = 'PORT'
            self.setup_error = "Connection Failed. Saloon is closed."

            lbl_color = SVGA['INK'] if self.setup_sequence.index(self.setup_step) >= idx else (180, 160, 130)
            self.screen.blit(self.ledger_font.render(label, True, lbl_color), (fx + 50, y_pos)) 
            
            box_x = fx + 420 
            if self.setup_step == step_id:
                pygame.draw.line(self.screen, SVGA['INK'], (box_x, y_pos + 25), (box_x + 150, y_pos + 25), 2)
                display_input = self.input_text + ("_" if self.blink_timer % 30 < 15 else "")
                self.screen.blit(self.ledger_font.render(display_input, True, SVGA['INK']), (box_x + 5, y_pos))
            else:
                if static_val: self.screen.blit(self.ledger_font.render(static_val, True, SVGA['INK']), (box_x + 5, y_pos))

        if self.setup_error:
            e_surf = self.ledger_font.render(self.setup_error, True, SVGA['TEXT_RED'])
            self.screen.blit(e_surf, (self.res[0]//2 - e_surf.get_width()//2, fy + form_h - 50))

    def draw_title_screen(self):

        cx = self.res[0] // 2

        self.draw_pistol(cx - 280, 90, flip_x=False)
        self.draw_pistol(cx + 280, 90, flip_x=True)
        
        rules = ["Goal: Collect the most cards, partner.", "A 'Bullseye' card is dealt face-up.", "When dealer yells 'CLICK CLICK', cards flip.", "QUICK DRAW: If ANY match Bullseye, press 'T' immediately!", "No match? Highest card takes the pot.", "TIE-BREAKER: Trigger a sudden-death Quickdraw round!", "MISFIRE: Press 'T' with no match? You lose the round!", "MOONSHINE ROUND: Zeros can steal from leaders at the end!"]
        y = 250
        for r in rules:
            self.screen.blit(self.font.render(r, True, SVGA['TEXT_LIGHT']), (cx - 400, y))

        self.blink_timer += 1
        if self.blink_timer % 30 < 15:
            prompt = self.font.render("PRESS SPACE TO KICK OPEN THE DOORS", True, SVGA['TEXT_BLUE'])
            self.screen.blit(prompt, (cx - prompt.get_width()//2, self.res[1] - 80))
        pygame.display.flip()

    def draw_door_anim_screen(self):
        self.draw_game_screen() 
        door_color, dark_wood = (120, 80, 30), (70, 40, 15)
        left_w = int((self.res[0] // 2) * (1.0 - self.door_progress))
        if self.door_progress < 1.0:
            pygame.draw.rect(self.screen, door_color, (0, 0, left_w, self.res[1]))
            pygame.draw.rect(self.screen, door_color, (self.res[0] - left_w, 0, left_w, self.res[1]))
        self.door_progress += 0.02
        if self.door_progress >= 1.0: self.state = 'GAME'
        pygame.display.flip()

    def draw_pistol(self, x, y, flip_x=False):
        surf = pygame.Surface((120, 70), pygame.SRCALPHA)
        pygame.draw.rect(surf, (150, 150, 160), (50, 15, 65, 12), border_radius=2)
        pygame.draw.rect(surf, SVGA['WOOD_BASE'], [(10, 60), (30, 60), (35, 30), (15, 30)])
        if flip_x: surf = pygame.transform.flip(surf, True, False)
        self.screen.blit(surf, (x - 60, y - 35))

    def draw_quickdraw_waiting_sign(self):
        cx, cy = self.left_panel_w + (self.res[0] - self.left_panel_w) // 2, self.res[1] // 2
        pygame.draw.rect(self.screen, SVGA['WOOD_BASE'], (cx-225, cy-100, 450, 200), border_radius=10)
        self.screen.blit(self.font.render("QUICKDRAW TIE-BREAKER!", True, SVGA['GOLD']), (cx - self.font.render("QUICKDRAW TIE-BREAKER!", True, SVGA['GOLD']).get_width()//2, cy - 80))

    def draw_moonshine_waiting_sign(self):
        cx, cy = self.left_panel_w + (self.res[0] - self.left_panel_w) // 2, self.res[1] // 2
        pygame.draw.rect(self.screen, SVGA['WOOD_BASE'], (cx-225, cy-140, 450, 280), border_radius=10)
        self.screen.blit(self.font.render("MOONSHINE SHOOTOUT!", True, SVGA['GOLD']), (cx - self.font.render("MOONSHINE SHOOTOUT!", True, SVGA['GOLD']).get_width()//2, cy - 120))

    def draw_moonshine_jug(self):
        cx, cy = self.left_panel_w + (self.res[0] - self.left_panel_w)//2, self.jug_y
        self.jug_y += self.jug_dy
        self.jug_dy += self.jug_gravity
        pygame.draw.circle(self.screen, (160, 110, 70), (cx, int(cy) + 30), 65)
        xxx_surf = self.font.render("XXX", True, (150, 20, 20))
        self.screen.blit(xxx_surf, (cx - xxx_surf.get_width()//2, int(cy) + 20))
        for bx, by in self.jug_bullet_holes:
            pygame.draw.circle(self.screen, (15, 15, 15), (cx + bx, int(cy) + by), 7)

    def draw_card(self, x, y, width, height, value, suit, label, revealed, is_winner=False, bullet_holes=None, has_tombstone=False, flash_alpha=0):
        if bullet_holes is None: bullet_holes = []
        self.screen.blit(self.small_font.render(label, True, SVGA['GOLD']), (x + width//2 - self.small_font.render(label, True, SVGA['GOLD']).get_width()//2, y - 25))
        pygame.draw.rect(self.screen, SVGA['WHITE'] if revealed else (20, 50, 100), (x, y, width, height), border_radius=8)
        if revealed:
            color = SVGA['SUIT_RED'] if suit in ['Hearts', 'Diamonds'] else SVGA['SUIT_BLACK']
            self.screen.blit(self.card_font.render(value, True, color), (x + 10, y + 10))
        for bx, by in bullet_holes:
            pygame.draw.circle(self.screen, (15, 15, 15), (x + width//2 + bx, y + height//2 + by), 7)
        if has_tombstone:
            pygame.draw.circle(self.screen, (100, 100, 100), (x + width//2, y + height//2), 30)

    def draw_game_screen(self):
        self.screen.fill(SVGA['BG_GREEN'])
        pygame.draw.rect(self.screen, SVGA['UI_PANEL'], (0, 0, self.left_panel_w, self.res[1]))
        self.screen.blit(self.font.render(f"GUNSLINGER: {self.name} | THUMPER: {self.thumper}", True, SVGA['GOLD']), (20, 20))
        for msg in self.log_messages:
            msg.update()
            if 60 < msg.y < self.res[1]: self.screen.blit(self.font.render(msg.text, True, msg.color), (20, msg.y))
        if not self.game_over:
            if self.moonshine_waiting: self.draw_moonshine_waiting_sign()
            elif self.show_moonshine_jug: self.draw_moonshine_jug()
            else:
                if self.bullseye_anim_card:
                    c = self.bullseye_anim_card
                    c.update()
                    self.draw_card(c.draw_x, c.draw_y, 150, 210, c.value, c.suit, c.label, c.revealed, bullet_holes=c.bullet_holes)
                for p_name, c in self.player_cards.items():
                    c.update()
                    self.draw_card(c.draw_x, c.draw_y, 100, 140, c.value, c.suit, "YOU" if p_name == self.name else p_name, c.revealed, c.is_winner, c.bullet_holes, c.has_tombstone, c.flash_alpha)
        pygame.display.flip()

    def process_messages(self):
        while not self.msg_queue.empty():
            text, color = self.msg_queue.get()
            self.add_log_message(text, color)

    def run(self):
        clock = pygame.time.Clock()
        while self.running:
            self.process_messages()
            for event in pygame.event.get():
                if event.type == pygame.QUIT: self.running = False
                elif event.type == pygame.KEYDOWN:
                    if self.state == 'SETUP': self.handle_setup_input(event)
                    elif self.state == 'TITLE' and event.key == pygame.K_SPACE:
                        self.state = 'DOOR_ANIM'
                        self.door_progress = 0.0
                    elif self.state == 'GAME':
                        if event.key == pygame.K_t:
                            if self.show_moonshine_jug and self.jug_y < self.res[1] + 150:
                                self.jug_bullet_holes.append((random.randint(-40, 40), random.randint(-40, 60)))
                                self.jug_shot = True
                            self.client_socket.sendall(json.dumps({'action': 'trigger'}).encode() + b"\n")
                        elif event.key == pygame.K_s and self.is_host:
                            self.client_socket.sendall(json.dumps({'action': 'start'}).encode() + b"\n")
            if self.state == 'SETUP': self.draw_setup_screen()
            elif self.state == 'TITLE': self.draw_title_screen()
            elif self.state == 'DOOR_ANIM': self.draw_door_anim_screen()
            elif self.state == 'GAME': self.draw_game_screen()
            clock.tick(60)
        pygame.quit()
        os._exit(0)

if __name__ == "__main__":
    TriggeredGameApp().run()
