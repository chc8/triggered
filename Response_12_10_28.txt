* **Connection Timeout:** If you are playing as a **Client (C)** and the IP address or Port is unreachable, the `client_socket.connect` call might hang the main thread, making the window appear to "crash" or become unresponsive before finally closing.

class TriggeredServer:

    def accept_clients(self):

            except:

    def safe_sleep(self, duration):

                    else:
                        self.broadcast({'msg': f"  [!] *BANG* {player_name} reached for their iron!"})
            elif self.game_active:
                self.broadcast({'msg': f"  [-] *Click* {player_name} drew too early!"})

    def handle_client(self, conn):
        try:
            data = conn.recv(1024).decode()
            if data:
                player_info = json.loads(data)
                player_info['conn'] = conn
                player_info['is_ai'] = False
                player_info['score'] = 0
                with self.lock:
                    self.clients[conn] = player_info
                    self.players.append(player_info)
                self.broadcast({'msg': f"Gunslinger {player_info['name']} moseyed into the saloon!"})
                
            buffer = ""
            while True:
                data = conn.recv(1024).decode()
                if not data: break
                buffer += data
                while "\n" in buffer:
                    line, buffer = buffer.split("\n", 1)
                    try:
                        parsed = json.loads(line.strip())
                        action = parsed.get('action')
                        if action == 'trigger' and self.game_active and not self.is_paused:
                            self.register_trigger(player_info['name'])
                        elif action == 'start' and not self.game_active:
                            threading.Thread(target=self.start_game_logic, daemon=True).start()
                    except: pass 
        except: pass
        finally:
            conn.close()

    def broadcast(self, message):
        msg_str = json.dumps(message) + "\n"
        with self.lock:
            for conn in list(self.clients.keys()):
                try: conn.sendall(msg_str.encode())
                except: pass

    def start_game_logic(self):
        with self.lock:
            if self.game_active: return

    def game_loop(self):
        deck = create_deck()
        random.shuffle(deck)
