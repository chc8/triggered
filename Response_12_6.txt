# --- Network Client (Graphical) ---
class PygameTriggeredClient:
    def __init__(self, host, port, name, thumper):
        self.host = host
        self.port = port
        self.name = name
        self.thumper = thumper
        self.client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.msg_queue = queue.Queue()
        self.running = True
        
        # State for drawing cards
        self.bullseye_card = None
        self.my_card = None
        self.cards_dealt = False
        
        # Pygame setup
        pygame.init()
        self.scale = 2
        self.res = (320 * self.scale, 200 * self.scale)
        self.screen = pygame.display.set_mode(self.res)
        pygame.display.set_caption("TRIGGERED - EGA Edition")
        
        # Font and word-wrapping setup
        self.font = pygame.font.SysFont('courier', 12 * self.scale, bold=True)
        char_width = self.font.size("A")[0]
        # Restrict text log to the left 420 pixels to leave room for cards on the right
        self.max_chars = 420 // char_width 
        
        self.log = [
            ("Welcome to the Saloon!", EGA['LIGHT_GRAY']),
            ("Press 'S' to Start game (if Host)", EGA['LIGHT_GRAY']),
            ("Press 'T' to pull the TRIGGER!", EGA['LIGHT_GRAY'])
        ]

    def connect(self):
        try:
            self.client_socket.connect((self.host, self.port))
            info = json.dumps({'name': self.name, 'thumper': self.thumper})
            self.client_socket.sendall(info.encode())
            self.msg_queue.put(("Connected to the server!", EGA['LIGHT_GRAY']))
        except ConnectionRefusedError:
            print("Couldn't find the saloon. Make sure the Host has started the server.")
            sys.exit()
            
        threading.Thread(target=self.receive_messages, daemon=True).start()

    def receive_messages(self):
        buffer = ""
        while self.running:
            try:
                data = self.client_socket.recv(2048).decode()
                if not data: break
                buffer += data
                
                while "\n" in buffer:
                    line, buffer = buffer.split("\n", 1)
                    line = line.strip()
                    if not line: continue
                    
                    try:
                        msg = json.loads(line)
                        if 'msg' in msg:
                            # --- Card Parsing Logic ---
                            raw_msg = msg['msg']
                            if "is shufflin'" in raw_msg:
                                self.bullseye_card = None
                                self.my_card = None
                                self.cards_dealt = False
                            elif "Cards dealt face down" in raw_msg:
                                self.cards_dealt = True
                            elif ">>>" in raw_msg and "of" in raw_msg:
                                parts = raw_msg.replace(">>>", "").replace("<<<", "").strip().split(" of ")
                                if len(parts) == 2:
                                    self.bullseye_card = (parts[0].strip(), parts[1].strip())
                            elif f"- {self.name}:" in raw_msg:
                                parts = raw_msg.split(f"- {self.name}: ")[1].strip().split(" of ")
                                if len(parts) == 2:
                                    self.my_card = (parts[0].strip(), parts[1].strip())
                            
                            # --- Text Rendering Logic ---
                            for m_line in raw_msg.split('\n'):
                                if m_line.strip() == "":
                                    self.msg_queue.put(("", EGA['LIGHT_GRAY'])) 
                                else:
                                    color = EGA['LIGHT_GRAY']
                                    if '>>>' in m_line: color = EGA['LIGHT_CYAN']
                                    elif 'BANG' in m_line or 'Misfire' in m_line: color = EGA['LIGHT_RED']
                                    elif 'wins' in m_line.lower(): color = EGA['LIGHT_GREEN']
                                    
                                    wrapped = textwrap.wrap(m_line, self.max_chars)
                                    for w_line in wrapped:
                                        self.msg_queue.put((w_line, color))
                                        
                        if msg.get('action') == 'quit':
                            self.running = False
                    except Exception:
                        pass
            except:
                self.msg_queue.put(("Disconnected from the server.", EGA['LIGHT_GRAY']))
                self.running = False
                break

    def draw_card_back(self, x, y, label):
        lbl_surf = self.font.render(label, True, EGA['YELLOW'])
        self.screen.blit(lbl_surf, (x, y - 25))
        
        w, h = 100, 140
        pygame.draw.rect(self.screen, EGA['BLUE'], (x, y, w, h))
        pygame.draw.rect(self.screen, EGA['WHITE'], (x, y, w, h), 4) # Border
        
        # Draw EGA crosshatch pattern
        for i in range(10, w, 20):
            pygame.draw.line(self.screen, EGA['CYAN'], (x+i, y), (x+i, y+h), 2)
        for i in range(10, h, 20):
            pygame.draw.line(self.screen, EGA['CYAN'], (x, y+i), (x+w, y+i), 2)

    def draw_card(self, x, y, value, suit, label):
        lbl_surf = self.font.render(label, True, EGA['YELLOW'])
        self.screen.blit(lbl_surf, (x, y - 25))
        
        w, h = 100, 140
        pygame.draw.rect(self.screen, EGA['WHITE'], (x, y, w, h))
        pygame.draw.rect(self.screen, EGA['LIGHT_GRAY'], (x, y, w, h), 4) # Border
        
        color = EGA['LIGHT_RED'] if suit in ['Hearts', 'Diamonds'] else EGA['BLACK']
        
        # Card Value
        val_surf = self.font.render(value, True, color)
        self.screen.blit(val_surf, (x + 10, y + 10))
        
        # Draw Pixel-Art Suit in the center
        cx = x + w // 2
        cy = y + h // 2
        
        if suit == 'Diamonds':
            pygame.draw.polygon(self.screen, color, [(cx, cy-20), (cx+20, cy), (cx, cy+20), (cx-20, cy)])
        elif suit == 'Hearts':
            pygame.draw.circle(self.screen, color, (cx-10, cy-10), 12)
            pygame.draw.circle(self.screen, color, (cx+10, cy-10), 12)
            pygame.draw.polygon(self.screen, color, [(cx-20, cy-5), (cx+20, cy-5), (cx, cy+20)])
        elif suit == 'Spades':
            pygame.draw.circle(self.screen, color, (cx-10, cy+5), 12)
            pygame.draw.circle(self.screen, color, (cx+10, cy+5), 12)
            pygame.draw.polygon(self.screen, color, [(cx-20, cy+5), (cx+20, cy+5), (cx, cy-20)])
            pygame.draw.rect(self.screen, color, (cx-4, cy+5, 8, 20))
        elif suit == 'Clubs':
            pygame.draw.circle(self.screen, color, (cx, cy-15), 12)
            pygame.draw.circle(self.screen, color, (cx-12, cy+5), 12)
            pygame.draw.circle(self.screen, color, (cx+12, cy+5), 12)
            pygame.draw.rect(self.screen, color, (cx-4, cy+5, 8, 20))

    def draw_screen(self):
        self.screen.fill(EGA['BLACK'])
        
        # Draw top banner
        banner = self.font.render(f"GUNSLINGER: {self.name} | THUMPER: {self.thumper}", True, EGA['YELLOW'])
        self.screen.blit(banner, (10, 10))
        pygame.draw.line(self.screen, EGA['BROWN'], (10, 35), (self.res[0]-10, 35), 2)
        
        # Process new messages
        while not self.msg_queue.empty():
            self.log.append(self.msg_queue.get())
            if len(self.log) > 15: # Keep log trimmed to screen height
                self.log.pop(0)
                
        # Draw game text log
        y = 50
        for item in self.log:
            line, color = item
            if line == "": 
                y += 20 * (self.scale // 2)
                continue
            text_surface = self.font.render(line, True, color)
            self.screen.blit(text_surface, (10, y))
            y += 20 * (self.scale // 2)

        # Draw Cards on the right side
        if self.bullseye_card:
            self.draw_card(480, 60, *self.bullseye_card, "BULLSEYE")
            
        if self.my_card:
            self.draw_card(480, 240, *self.my_card, "YOUR CARD")
        elif self.cards_dealt:
            self.draw_card_back(480, 240, "YOUR CARD")

        pygame.display.flip()

    def start(self):
        self.connect()
        clock = pygame.time.Clock()
        
        while self.running:
            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    self.running = False
                    msg = json.dumps({'action': 'exit'}) + "\n"
                    try:
                        self.client_socket.sendall(msg.encode())
                    except:
                        pass
                elif event.type == pygame.KEYDOWN:
                    if event.key == pygame.K_t: # QUICK DRAW!
                        self.msg_queue.put((">>> You reached for your iron! <<<", EGA['LIGHT_CYAN']))
                        msg = json.dumps({'action': 'trigger'}) + "\n"
                        self.client_socket.sendall(msg.encode())
                    elif event.key == pygame.K_s: # START GAME
                        msg = json.dumps({'action': 'start'}) + "\n"
                        self.client_socket.sendall(msg.encode())
                    elif event.key == pygame.K_r: # RESTART GAME
                        msg = json.dumps({'action': 'restart'}) + "\n"
                        self.client_socket.sendall(msg.encode())

            self.draw_screen()
            clock.tick(30) # 30 FPS is plenty for an EGA feel
            
        pygame.quit()
        os._exit(0)
